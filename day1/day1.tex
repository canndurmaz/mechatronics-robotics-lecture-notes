\begin{columns}[T]
\begin{column}{.5\textwidth}
\begin{defbox}[Refresher: power equals torque times speed]
\small
Linear: $P=F\,v$ \quad Rotation: $P=T\,\omega$\\
$T$ in \si{\newton\metre}, $\omega$ in \si{\radian\per\second}, $P$ in \si{\watt};\; $\omega=\dfrac{2\pi n}{60}$ ($n$ in rpm)\\
\textbf{Example:} \SI{1.27}{\newton\metre} at \SI{3000}{rpm}: $\omega=\SI{314}{\radian\per\second}$ $\Rightarrow$ $P\approx\SI{400}{\watt}$.
\end{defbox}
\end{column}
\begin{column}{.46\textwidth}
\small
\begin{itemize}\setlength\itemsep{1pt}
\item Motor: electrical power $v\,i$ in, mechanical power $T\,\omega$ out.
\item Efficiency $\eta=T\omega/(v\,i)\approx 85$--$95\,\%$.
\item Losses become \hl{heat}: heat limits the motor.
\item Works both ways: braking turns the motor into a \term{generator}.
\end{itemize}
\end{column}
\end{columns}

\begin{frame}{Refresher: magnetic field and the force on a wire}
\begin{columns}[T]
\begin{column}{.43\textwidth}
\centering
\begin{tikzpicture}[scale=.95]
\fill[mkRed!70] (0,2.4) rectangle (4,2.9); \node[white,font=\scriptsize\bfseries] at (2,2.65) {N};

\draw[mkGray,line width=1pt] (0.2,0.3) -- (3.9,0.3); \draw[mkGray,line width=1pt] (0.2,1.7) -- (3.9,1.7);
\fill[mkOrange] (2.1,0.2) rectangle (2.35,1.8);
\draw[-{Stealth},mkOrange,line width=1pt] (2.22,0.45) -- (2.22,1.55);
\node[font=\scriptsize,text=mkOrange,right] at (2.35,1.25) {$i$};
\draw[-{Stealth},mkNavy,line width=1.4pt] (2.5,1.0) -- (3.5,1.0) node[right,font=\scriptsize]{$F$};
\draw[|-|,mkNavy] (1.8,0.3) -- node[left,font=\scriptsize]{$l$} (1.8,1.7);
\node[font=\scriptsize] at (2,-1.2) {\color{mkGray}wire on two rails between magnet poles};
\end{tikzpicture}
\end{column}
\begin{column}{.55\textwidth}
\small
\begin{itemize}\setlength\itemsep{2pt}
\item A magnet fills space with a \term{magnetic field} $\bm{B}$ (unit: tesla, \si{\tesla}). Picture it as arrows from N to S.
\item A wire carrying current through that field is \hl{pushed sideways} -- perpendicular to both $i$ and $\bm{B}$.
\item Direction: right-hand rule (fingers along $i$, bend toward $\bm{B}$, thumb $=$ force).
\end{itemize}
\begin{eqbox}
$F = B\,l\,i$ \qquad $F$ [\si{\newton}], $B$ [\si{\tesla}], $l$ [\si{\metre}], $i$ [\si{\ampere}]
\end{eqbox}
\textbf{Example:} $B=\SI{0.5}{\tesla}$, $l=\SI{0.1}{\metre}$, $i=\SI{4}{\ampere}$ $\Rightarrow$ $F=\SI{0.2}{\newton}$.\\[1mm]
\hl{Take-away:} current $+$ magnetic field $=$ force. Every motor uses this.
\end{column}
\end{columns}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Lorentz force and Faraday's law: the two halves of every motor}
\begin{columns}[T]
\begin{column}{.47\textwidth}
\small
\textbf{1. Motor action (Lorentz):} current in a field gives force, $F=Bli$.\\[1mm]
\textbf{2. Generator action (Faraday):} a wire \hl{moving} at speed $v$ through the field gets a voltage $e=Blv$, the \term{back-EMF}.\\[1mm]
Both happen at the same time in every motor.\\[2mm]
Bend the wire into a coil ($N$ turns, radius $r$) and both laws use the same constant $k=2NBlr$:
\begin{eqbox}
$T=k_t\,i$, \qquad $e=k_e\,\omega$, \qquad $k_t=k_e$ (SI units)
\end{eqbox}
\end{column}
\begin{column}{.5\textwidth}
\begin{defbox}[Worked example]
\small
Coil: $N=50$, $B=\SI{0.5}{\tesla}$, $l=\SI{0.04}{\metre}$, $r=\SI{0.02}{\metre}$
\begin{enumerate}\setlength\itemsep{1pt}
\item $k=2\cdot50\cdot0.5\cdot0.04\cdot0.02=\SI{0.04}{\newton\metre\per\ampere}$
\item At $i=\SI{5}{\ampere}$: $T=0.04\cdot5=\SI{0.2}{\newton\metre}$
\item At $\omega=\SI{300}{\radian\per\second}$: $e=0.04\cdot300=\SI{12}{\volt}$
\end{enumerate}
\end{defbox}
\begin{keybox}[Take-away]
\small
\term{Torque constant} $k_t$ (\si{\newton\metre\per\ampere}) and \term{back-EMF constant} $k_e$ (\si{\volt\second\per\radian}) are the same number: \hl{current makes torque, speed makes voltage}.
\end{keybox}
\end{column}
\end{columns}
\end{frame}

\begin{columns}[T]

\begin{tikzpicture}[scale=.8]

\begin{tikzpicture}
\begin{axis}[mkplot,width=5.2cm,height=3.4cm,xmin=0,xmax=180,xtick={0,45,90,135,180},

\end{axis}

\end{column}
\begin{column}{.57\textwidth}
\small
\textbf{Analogy:} a fridge magnet snaps onto the steel door -- the iron moves to where the magnetic path is \hl{easiest}.
\begin{itemize}\setlength\itemsep{2pt}
\item No magnet needed on the rotor: a plain iron bar turns to line up with the coil.
\item Torque pushes towards the position of \hl{highest inductance} $L$.
\item Torque grows with $i^2$, so the current direction does not matter.
\item Used in solenoids, \term{stepper motors} and switched-reluctance motors.
\end{itemize}
\begin{eqbox}
$T=\tfrac12\,i^2\,\dfrac{dL}{d\theta}$ \quad ($L$ in \si{\henry}, $\theta$ in rad)
\end{eqbox}
\gofurther{the general co-energy method $T=\partial W'/\partial\theta$ also covers saturated iron and the extra ``reluctance torque'' of interior-magnet motors [Fitzgerald et al., 2003, ch.\,3].}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Session 1.1 -- Summary and key terms}
\begin{columns}[T]
\begin{column}{.55\textwidth}
\small
\begin{itemize}
\item A motor turns $v\,i$ into $T\,\omega$ through a magnetic field -- and back again when braking.
\item Current in a field gives force ($F=Bli$); motion in a field gives voltage ($e=Blv$).
\item Iron alone is also pulled into a field (reluctance).
\item \hl{Current limits torque, voltage limits speed, heat limits continuous use.}
\end{itemize}
\begin{keybox}[Remember]
$P=T\omega$, \quad $T=k_t\,i$, \quad $e=k_e\,\omega$, \quad $k_t=k_e$
\end{keybox}
\end{column}
\begin{column}{.42\textwidth}
\begin{defbox}[Key terms]
\small
\term{Magnetic field $B$}: invisible ``push field'' of a magnet, in tesla.\\
\term{Back-EMF}: voltage a spinning motor generates.\\
\term{Torque constant $k_t$}: torque per ampere.\\
\term{Reluctance}: ``difficulty'' of the magnetic path.\\
\term{Efficiency $\eta$}: power out / power in.
\end{defbox}
\end{column}
\end{columns}
\end{frame}

\end{itemize}
\begin{keybox}[Idea for brushless motors]
Replace the mechanical switch by \hl{electronic switches + a rotor-angle sensor}.
\end{keybox}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Induction motor and stepper motor: short introduction}
\begin{columns}[T]
\begin{column}{.22\textwidth}
\centering
\figph{width=\linewidth,height=2.6cm,keepaspectratio}{d1_squirrel_cage.jpg}\\
\credit{Zureks, CC BY-SA 3.0, via Wikimedia Commons}\\[1mm]
\figph{width=\linewidth,height=2.6cm,keepaspectratio}{d1_stepper.jpg}\\
\credit{tony\_duell, CC BY 2.0, via Wikimedia Commons}
\end{column}
\begin{column}{.38\textwidth}
\small
\textbf{Induction motor} (squirrel-cage rotor)
\begin{itemize}\setlength\itemsep{1pt}
\item Rotating stator field \hl{induces} currents in the rotor bars -- no magnets.
\item Rotor runs slightly slower than the field: \term{slip} $s$ (a few \%).
\item Cheap, robust: pumps, fans, big spindles.
\end{itemize}
\begin{eqbox}
$n_s=\dfrac{60f}{p}$, \quad $s=\dfrac{n_s-n}{n_s}$
\end{eqbox}
{\small Example: $f=\SI{50}{\hertz}$, $p=2$: $n_s=\SI{1500}{rpm}$; at \SI{1440}{rpm}, $s=4\,\%$.}
\end{column}
\begin{column}{.38\textwidth}
\small
\textbf{Stepper motor} (reluctance + magnet)
\begin{itemize}\setlength\itemsep{1pt}
\item Each current pulse moves the rotor one \term{step}, e.g.\ 1.8$^\circ$ (200 steps/rev).
\item Position by \hl{counting steps}: no sensor needed (\term{open loop}).
\item Risk: if the load is too heavy, steps are \hl{lost} and nobody notices.
\item Torque drops quickly with speed.
\end{itemize}
\gofurther{torque--slip curve (Kloss formula), vector control of induction motors, micro-stepping and stepper resonance.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{The two motor equations}
\begin{columns}[T]
\begin{column}{.52\textwidth}
\small
\textbf{Electrical (KVL around the loop):}
\[
v = R\,i + L\frac{di}{dt} + k_e\,\omega
\]
\textbf{Mechanical (Newton for rotation):}
\begin{eqbox}
$J\dfrac{d\omega}{dt} = k_t\,i - B\,\omega - T_L$
\end{eqbox}
\begin{itemize}\setlength\itemsep{1pt}
\item Linked both ways: $i$ makes torque, $\omega$ makes back-EMF.
\item Two first-order ODEs $\Rightarrow$ one \hl{second-order} system.
\end{itemize}
\end{column}
\begin{column}{.45\textwidth}
\begin{keybox}[Why k\textsubscript{t} equals k\textsubscript{e}: energy conservation]
\small
Power converted on the electrical side: $e\,i=k_e\,\omega\,i$\\
Power delivered on the mechanical side: $T\,\omega=k_t\,i\,\omega$\\
They must be equal $\Rightarrow$ $\bm{k_t=k_e}$ (SI units).
\end{keybox}
\small
\textbf{Datasheet units:} $k_e$ is often given in V per 1000 rpm.\\
Divide by 104.7 to get \si{\volt\second\per\radian}:\\
\SI{10.47}{\volt}/krpm $\Rightarrow$ $k_e=k_t=0.1$ SI.\\[1mm]
\gofurther{three-phase datasheets use rms and line-to-line values, giving $k_t=\sqrt3\,k_e$; resistance rises 0.4\,\%/K when hot.}
\end{column}
\end{columns}
\end{frame}

\begin{column}{.48\textwidth}

\end{eqbox}

\item After $1\tau$: 63\,\%; after $3\tau$: 95\,\%; after $5\tau$: ``done''.
\end{itemize}

\end{column}
\begin{column}{.43\textwidth}
\small
\begin{enumerate}\setlength\itemsep{2pt}
\item Electrical: $\tau_e=L/R=\SI{0.5}{\milli\second}$.
\item Mechanical: $\tau_m=\dfrac{RJ}{k_tk_e}=\dfrac{1\cdot10^{-4}}{0.01}=\SI{10}{\milli\second}$.
\item Final speed: $\omega_\infty\approx v/k_e=\SI{240}{\radian\per\second}$.
\item Start current: $\omega=0$, no back-EMF $\Rightarrow$ up to $v/R=\SI{24}{\ampere}$ (peak \SI{21}{\ampere}).
\end{enumerate}
\begin{eqbox}
$\tau_e=\dfrac{L}{R}$ \quad $\ll$ \quad $\tau_m=\dfrac{RJ}{k_tk_e}$
\end{eqbox}
\hl{Take-away:} current is fast, speed is slow; more load inertia $\Rightarrow$ larger $\tau_m$. Real drives limit the start current.
\end{column}

\begin{frame}{Session 1.3 -- Summary and key terms}
\begin{columns}[T]
\begin{column}{.55\textwidth}
\small
\begin{itemize}
\item Motor = RL circuit + back-EMF, coupled to an inertia with friction and load.
\item Energy conservation gives $k_t=k_e$ (SI units).
\item Constant speed: straight speed--torque line.
\item Two time constants: fast electrical $\tau_e$, slow mechanical $\tau_m$.

\end{keybox}

\term{Moment of inertia $J$}: resistance to changes of rotation speed.\\
\term{Time constant $\tau$}: time to reach 63\,\%.\\
\term{Stiction}: sticking friction at standstill.\\
\term{Block diagram}: picture of equations as signal flow.\\
\term{Disturbance}: unwanted input such as $T_L$.
\end{defbox}

\item Motor \SI{1}{\newton\metre} $\Rightarrow$ load $3\cdot1=\SI{3}{\newton\metre}$.

\begin{frame}{Reflected inertia: what the motor ``feels''}
\begin{columns}[T]
\begin{column}{.44\textwidth}
\centering
\begin{tikzpicture}[scale=.9]

\draw[-{Stealth},mkRed] (4.1,1.0) arc (60:-60:0.45) node[right,font=\scriptsize,text=mkRed] {$\dfrac{T_L}{N}$};
\node[font=\scriptsize,text=mkGray] at (1.8,-0.35) {equivalent load at the motor shaft};
\end{tikzpicture}

\vspace{3mm}
\small
\textbf{Bicycle again:} in a low gear the bike feels \hl{light} to accelerate -- but you must pedal fast.
\end{column}
\begin{column}{.53\textwidth}
\small
Kinetic energy must be the same on both sides:
$\tfrac12 J_L\omega_L^2=\tfrac12\,\dfrac{J_L}{N^2}\,\omega_m^2$ \; (since $\omega_L=\omega_m/N$)
\begin{eqbox}
$J_\text{tot}=J_m+\dfrac{J_L}{N^2}$, \qquad load torque seen: $\dfrac{T_L}{N}$
\end{eqbox}
\textbf{Why $N^2$?} The load turns $N\times$ slower (one factor) \emph{and} the gear multiplies torque by $N$ (second factor).\\[1mm]
\textbf{Example:} $J_L=\SI{9e-4}{\kilo\gram\metre\squared}$, $N=3$ $\Rightarrow$ $J_L/N^2=\SI{1e-4}{\kilo\gram\metre\squared}$.\\[1mm]

\toprule
\textbf{Mechanism} & \textbf{Inertia at motor}\\
\midrule
Gear, ratio $N$ & $J_L/N^2$\\
Ball screw, lead $p$ & $m\,(p/2\pi)^2$\\
Belt / rack, radius $r$ & $m\,r^2$\\
\bottomrule
\end{tabular}
\end{column}

\begin{frame}{Session 1.4 -- Summary and key terms}
\begin{columns}[T]
\begin{column}{.55\textwidth}
\small
\begin{itemize}
\item Gears and screws trade speed for torque/force.
\item The load is ``reflected'' to the motor shaft by \hl{energy equivalence}: inertia $\div N^2$.
\item Best acceleration: $N_\text{opt}=\sqrt{J_L/J_m}$; speed limits often decide.
\item Linear motors and voice coils: same equations with $F,v,m$.
\end{itemize}
\begin{keybox}[Remember]
$N=\omega_m/\omega_L$, \; $J_L'=J_L/N^2$, \; $v=\dfrac{p}{2\pi}\omega$, \; $J'_\text{screw}=m\left(\dfrac{p}{2\pi}\right)^2$
\end{keybox}
\end{column}
\begin{column}{.42\textwidth}
\begin{defbox}[Key terms]
\small
\term{Gear ratio $N$}: motor speed / load speed.\\
\term{Lead $p$}: travel per screw revolution.\\

\end{defbox}

\begin{column}{.46\textwidth}
\small
Heat $\propto T^2$, and the motor heats up slowly, so it ``feels'' the \hl{average of $T^2$} over the cycle:
\begin{eqbox}
$T_\text{rms}=\sqrt{\dfrac{1}{t_\text{cyc}}\displaystyle\sum_k T_k^2\,t_k}\;\le\;T_\text{cont}$
\end{eqbox}
\begin{itemize}\setlength\itemsep{1pt}
\item $T_k$: torque in segment $k$, lasting $t_k$ seconds.
\item Why not the plain average? Braking torque is negative but heats just as much.
\item Also check: largest $|T_k|\le T_\text{peak}$, max speed $\le n_\text{max}$.
\item Vertical axis: holding against gravity during ``wait'' also counts!
\end{itemize}

\begin{frame}{Motor--drive matching and selection checklist}
\begin{columns}[T]
\begin{column}{.5\textwidth}
\small
\textbf{Voltage:} at top speed the drive must beat the back-EMF:
\begin{eqbox}
$V_\text{supply}\;\gtrsim\;k_e\,\omega_\text{max}+R\,I_\text{peak}$
\end{eqbox}
\textbf{Current:} $I_\text{peak}=T_\text{peak}/k_t$ must be within the drive's rating.\\[1mm]
\textbf{Example} ($k_e=k_t=\SI{0.3}{}$ SI):
\begin{enumerate}\setlength\itemsep{1pt}
\item Back-EMF at \SI{314}{\radian\per\second}: $0.3\cdot314\approx\SI{94}{\volt}$.
\item A \SI{48}{\volt} drive is too small; a 230\,V drive (\SI{325}{\volt} DC) is fine.
\item $I_\text{peak}=0.92/0.3\approx\SI{3}{\ampere}$.
\end{enumerate}
\gofurther{rms/line-to-line datasheet conventions, inductive voltage drop, braking (regenerated) energy.}
\end{column}
\begin{column}{.47\textwidth}
\begin{keybox}[Selection checklist]
\small
\begin{enumerate}\setlength\itemsep{0pt}
\item Describe the move, the load and the cycle.
\item Reflect inertia and torques to the motor.
\item Choose the gear/screw ratio.
\item Peak torque inside the \hl{short-time zone}.
\item $T_\text{rms}$ inside the \hl{continuous zone} (+20--30\,\% margin).
\item Inertia ratio OK for the mechanics.
\item Drive: voltage and current.
\item Encoder, brake, protection class.
\end{enumerate}
\end{keybox}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Session 1.5 -- Summary and key terms}
\begin{columns}[T]
\begin{column}{.55\textwidth}
\small
\begin{itemize}
\item The torque--speed envelope has a continuous (thermal) zone and a short-time (current/voltage) zone.
\item The motor warms up like an RC circuit charges: exponentially, over minutes.
\item Judge heating with $T_\text{rms}$, acceleration with $T_\text{peak}$.
\item Size step by step: speed $\to$ inertia $\to$ torques $\to$ RMS/peak $\to$ drive.
\end{itemize}
\begin{keybox}[Remember]
$T_\text{rms}=\sqrt{\tfrac{1}{t_\text{cyc}}\sum T_k^2t_k}$, \quad $V\gtrsim k_e\omega_\text{max}+RI_\text{pk}$
\end{keybox}
\end{column}
\begin{column}{.42\textwidth}
\begin{defbox}[Key terms]
\small
\term{Continuous torque}: torque possible forever.\\
\term{Peak torque}: short-burst torque.\\
\term{RMS torque}: ``heating-equivalent'' torque of a cycle.\\
\term{Duty cycle}: repeating pattern of the move.\\
\term{Thermal resistance}: K of rise per W of loss.
\end{defbox}
\end{column}

\fill[mkSky!70,draw=mkBlue] (2.4,0) rectangle (2.9,2.4); \node[font=\scriptsize] at (2.65,2.65) {$J$};
\draw[mkGray,line width=1.5pt] (2.9,1.2) -- (3.6,1.2);
\fill[mkGray!40] (3.6,0.8) rectangle (4.0,1.6); \node[font=\scriptsize] at (3.8,0.55) {$B$ (bearing)};
\draw[-{Stealth},mkOrange,line width=1pt] (3.2,1.8) arc (100:260:0.25 and 0.6);
\node[font=\scriptsize,text=mkOrange] at (3.35,2.15) {$T(t)$};
\draw[-{Stealth},mkNavy] (2.65,-0.2) arc (-90:30:0.35) ;
\node[font=\scriptsize] at (3.2,-0.3) {$\theta$};
\end{tikzpicture}
\begin{eqbox}
$J\,\ddot\theta + B\,\dot\theta + k_\theta\,\theta = T(t)$
\end{eqbox}
\small
Exactly the mass--spring--damper with $m\to J$, $b\to B$, $k\to k_\theta$, $F\to T$.
\end{column}
\begin{column}{.52\textwidth}
\small
\begin{itemize}\setlength\itemsep{3pt}
\item Newton for rotation: $\sum T = J\ddot\theta$.
\item A shaft twists like a spring: torsional stiffness $k_\theta$ [\si{\newton\metre\per\radian}].
\item Solid shaft: $k_\theta=\dfrac{G\,\pi d^4}{32\,l}$ ($G$: shear modulus).
\item Example: steel ($G=\SI{80}{\giga\pascal}$), $d=\SI{10}{\milli\metre}$, $l=\SI{100}{\milli\metre}$ $\Rightarrow$ $k_\theta\approx\SI{785}{\newton\metre\per\radian}$.
\item Couplings, belts and gear teeth are springs too -- the \hl{softest one dominates}.
\item Motor without load: $k_\theta=0$, so only $J\ddot\theta+B\dot\theta=T$.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Electrical RLC network}
\begin{columns}[T]
\begin{column}{.46\textwidth}
\centering
\begin{circuitikz}[scale=.85,transform shape,font=\small]
\draw (0,0) to[V,v<=$v(t)$,color=mkNavy] (0,3)
 to[R,l=$R$,i>^=$i$,color=mkNavy] (3,3)
 to[L,l=$L$,color=mkNavy] (5,3)
 to[C,l=$C$,v^<=$v_C$,color=mkOrange] (5,0) -- (0,0);
\end{circuitikz}
\begin{eqbox}
$L\,\ddot q + R\,\dot q + \dfrac{1}{C}\,q = v(t)$, \quad $i=\dot q$
\end{eqbox}
\end{column}
\begin{column}{.52\textwidth}
\small
\begin{itemize}\setlength\itemsep{3pt}
\item KVL: $v=v_R+v_L+v_C$ with $v_R=Ri$, $v_L=L\,di/dt$, $v_C=q/C$.
\item Write it with the charge $q$: the \hl{same form} as $m\ddot x+b\dot x+kx=F$!
\item $L$ acts like mass, $R$ like damper, $1/C$ like spring.
\item $\omega_n=\dfrac{1}{\sqrt{LC}}$, \quad $\zeta=\dfrac{R}{2}\sqrt{\dfrac{C}{L}}$.
\item Example: $L=\SI{1}{\milli\henry}$, $C=\SI{470}{\micro\farad}$, $R=\SI{0.1}{\ohm}$ $\Rightarrow$ $f_n=\SI{232}{\hertz}$, $\zeta=0.034$: rings a lot.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Analogies: one equation, three domains -- and the motor links them}
\begin{columns}[T]
\begin{column}{.5\textwidth}
\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{Translation} & \textbf{Rotation} & \textbf{Electrical}\\
\midrule
force $F$ & torque $T$ & voltage $v$\\
velocity $\dot x$ & speed $\omega$ & current $i$\\

damper $b$ & damper $B$ & resistance $R$\\

\bottomrule
\end{tabular}\\[2mm]

\end{column}

\end{tikzpicture}}

\vspace{2mm}
\small
\begin{itemize}\setlength\itemsep{2pt}
\item The motor is the bridge: it converts power between the domains without storing any.
\item Two energy stores ($L$ and $J$) $\Rightarrow$ 2nd-order system, as in Session 1.3.
\end{itemize}
\gofurther{energy-based modelling with Lagrange's equations; ``gyrator'' in bond-graph language; state-space form $\dot{\bm x}=A\bm x+B\bm u$ (Day 2).}
\end{column}

\begin{frame}{Session 2.1 (part 1) -- Summary and key terms}
\begin{columns}[T]
\begin{column}{.55\textwidth}
\small
\begin{itemize}
\item Lumped models: masses, springs, dampers (and $L$, $C$, $R$).
\item Mass--spring--damper, shaft and RLC circuit share one 2nd-order ODE with $\omega_n$ and $\zeta$.
\item Low damping $\Rightarrow$ overshoot and ringing.
\item The motor links the electrical and mechanical worlds.
\item A springy shaft between motor and load creates resonance.
\end{itemize}
\begin{keybox}[Remember]
$m\ddot x+b\dot x+kx=F$, \quad $\omega_n=\sqrt{k/m}$, \quad $\zeta=\dfrac{b}{2\sqrt{km}}$
\end{keybox}
\end{column}
\begin{column}{.42\textwidth}
\begin{defbox}[Key terms]
\small
\term{Lumped model}: system built from ideal elements.\\
\term{Natural frequency $\omega_n$}: speed of free oscillation.\\
\term{Damping ratio $\zeta$}: how fast oscillation dies.\\
\term{Step response}: reaction to a sudden constant input.\\
\term{Resonance}: big response at one special frequency.
\end{defbox}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Day 1 summary}
\begin{columns}[T]
\begin{column}{.5\textwidth}
\small
\begin{defbox}[Machines]
Current $+$ field $=$ force; motion $+$ field $=$ voltage; $k_t=k_e$.\\
DC (brushes), BLDC/PMSM (electronic), induction (slip), stepper (steps).
\end{defbox}
\begin{defbox}[Models]
$v=Ri+L\dot i+k_e\omega$, \; $J\dot\omega=k_ti-B\omega-T_L$\\
Fast $\tau_e=L/R$, slow $\tau_m=RJ/(k_tk_e)$.
\end{defbox}
\end{column}
\begin{column}{.48\textwidth}
\small
\begin{defbox}[Transmissions and sizing]
Speed $\div N$, torque $\times N$, inertia $\div N^2$; $N_\text{opt}=\sqrt{J_L/J_m}$.\\
Peak torque in the short-time zone, $T_\text{rms}$ in the continuous zone.
\end{defbox}
\begin{defbox}[Dynamic modelling]
Lumped elements, analogies; 2nd-order system with $\omega_n,\zeta$; resonance.
\end{defbox}
\end{column}
\end{columns}
\vspace{2mm}
\begin{keybox}[One sentence]
A servo axis is an \hl{energy converter} (motor) plus a \hl{mechanical network} (transmission, load) -- both described by the same kind of ODEs.
\end{keybox}
\end{frame}

  \addlegendentry{Stribeck + Coulomb + viscous}

  \addplot[mkOrange,thick,dashed,domain=0.001:20] {30+1.2*x}; \addlegendentry{Coulomb + viscous}
  \addplot[mkOrange,thick,dashed,domain=-20:-0.001,forget plot] {-30+1.2*x};
  \addplot[mkSky,thick,dotted,domain=-20:20] {1.2*x}; \addlegendentry{viscous only}
  \draw[mkBlue,very thick] (axis cs:0,-50)--(axis cs:0,50);
  \node[font=\tiny,mkNavy,anchor=west] at (axis cs:0.5,53) {$F_S$};
  \node[font=\tiny,mkNavy,anchor=west] at (axis cs:7,28) {$F_C$};
\end{axis}
\end{tikzpicture}
\column{.48\textwidth}
\small
Think of pushing a heavy wardrobe: it needs a big push to start, then less to keep it sliding.
\begin{eqbox}
\[
F_f(v)=\underbrace{F_C\operatorname{sgn}(v)}_{\text{Coulomb}}+\underbrace{B\,v}_{\text{viscous}}
\]
\end{eqbox}
\footnotesize
\begin{itemize}
  \item $F_S$ breakaway force [N], $F_C$ Coulomb level [N], $B$ viscous coefficient [N\,s/m].
  \item At $v=0$ the force jumps: \hl{no slope} exists, so it cannot be linearised.
  \item Result in practice: \term{stick--slip} and ``hunting'' around the set-point.
  \item Fix: add the expected friction force as feedforward.
\end{itemize}
\further{Stribeck curve $F_C+(F_S-F_C)e^{-(v/v_S)^2}$, LuGre model.} \src{Armstrong-H\'elouvry et al., Automatica 1994}
\end{columns}
\end{frame}

\begin{frame}{Linearisation: zoom in until the curve looks straight}
\begin{columns}[T]
\column{.48\textwidth}
\centering
\begin{tikzpicture}
\begin{axis}[mkplot,width=6.6cm,height=5.2cm,xlabel={$x$},ylabel={$f(x)$},xmin=0,xmax=4,ymin=0,ymax=16]
  \addplot[mkBlue,very thick,domain=0:4,samples=60] {x^2};
  \addplot[mkOrange,thick,dashed,domain=1.6:4] {9+6*(x-3)};
  \fill[mkOrange] (axis cs:3,9) circle (2pt);
  \draw[mkGray,dashed] (axis cs:3,0)--(axis cs:3,9);
  \node[font=\tiny,anchor=north west] at (axis cs:3,0.9) {$x_0$};
  \node[font=\tiny,mkOrange,anchor=west] at (axis cs:2.05,13) {tangent, slope $f'(x_0)$};
  \node[font=\tiny,mkBlue,anchor=west] at (axis cs:0.3,3) {$f(x)=x^2$};
\end{axis}
\end{tikzpicture}
\column{.5\textwidth}
\small
Close to an \term{operating point} $x_0$ every smooth curve looks like its tangent line.
\begin{eqbox}
\[
f(x)\approx f(x_0)+f'(x_0)\,(x-x_0)
\]
\end{eqbox}
\footnotesize
\textbf{Example} $f=x^2$, $x_0=3$:
\begin{enumerate}
  \item slope $f'(3)=2\cdot3=6$
  \item at $x=3.1$: linear $9+6\cdot0.1=9.6$, exact $9.61$ (0.1\,\% error)
  \item at $x=4$: linear $15$, exact $16$ (6\,\% error) $\Rightarrow$ only \hl{local}
\end{enumerate}
$\delta x=x-x_0$ is the small \term{deviation} from the operating point.\\[2pt]
\further{many variables: $\delta\dot{\bm x}=A\,\delta\bm x+B\,\delta\bm u$ with Jacobians $A=\partial\bm f/\partial\bm x$, $B=\partial\bm f/\partial\bm u$; eigenvalues of $A$ give local stability (Lyapunov's indirect method).}
\end{columns}
\end{frame}

\column{.42\textwidth}
\footnotesize
$J\ddot\theta=\tau-mgl\sin\theta-b\dot\theta$ \quad($J$ [kg\,m$^2$], $\tau$ [Nm], $b$ [Nm\,s/rad])
\begin{enumerate}
  \item Only $\sin\theta$ is nonlinear; slope of $\sin$ at $\theta_0$ is $\cos\theta_0$.
  \item Linear model:
\end{enumerate}
\begin{eqbox}
\[
J\,\delta\ddot\theta+b\,\delta\dot\theta+(mgl\cos\theta_0)\,\delta\theta=\delta\tau
\]
\end{eqbox}
\begin{enumerate}\setcounter{enumi}{2}
  \item \textbf{Hanging} ($\cos0=+1$): gravity acts like a spring $\Rightarrow$ \hl{stable} swing.
  \item \textbf{Inverted} ($\cos\pi=-1$): a ``negative spring'' $\Rightarrow$ \hl{falls over}.
\end{enumerate}
Numbers: $m=2$\,kg, $l=0.3$\,m, $J=ml^2$ $\Rightarrow$ $\omega_n=\sqrt{g/l}=5.7$\,rad/s (0.9\,Hz).

  \node[font=\scriptsize] at (-1.4,1) {$i$};
\end{tikzpicture}\\[1mm]
\includegraphics[width=.9\linewidth]{../figures/d2_solenoid.jpg}\\
\credit{Svjo, CC BY-SA 3.0, via Wikimedia Commons}
\column{.72\textwidth}
\small
Magnet force $F=k\,i^2/z^2$ (current $i$ [A], air gap $z$ [m]). Like balancing a broom: the closer the iron, the \hl{stronger} the pull.
\begin{eqbox}
\[
\delta F=K_i\,\delta i-K_z\,\delta z\quad\Rightarrow\quad
\frac{\delta Z(s)}{\delta I(s)}=\frac{-K_i}{m s^2-K_z}
\]
\end{eqbox}
\footnotesize
\textbf{Worked example:} $m=0.1$\,kg, gap $z_0=5$\,mm, current $i_0=1$\,A.
\begin{enumerate}
  \item Weight: $mg=0.98$\,N must equal the magnet force at $(z_0,i_0)$.
  \item Slopes: $K_z=2mg/z_0=392$\,N/m (\hl{negative} stiffness), $K_i=2mg/i_0=1.96$\,N/A.
  \item Poles: $s=\pm\sqrt{K_z/m}=\pm62.6$\,s$^{-1}$ $\Rightarrow$ one pole in the right half-plane: \hl{unstable}.
  \item $K_z\propto1/z_0$: at another gap the model changes $\Rightarrow$ valid only near $(z_0,i_0)$.
\end{enumerate}
\end{columns}
\end{frame}

\begin{frame}{Session 2.1 -- Summary and key terms}
\begin{columns}[T]
\column{.55\textwidth}
\begin{keybox}[Key take-aways]
\footnotesize
\begin{itemize}
  \item Real axes have saturation, dead-zone, backlash, friction and magnetic saturation.
  \item Smooth curves are \hl{linearised} with the tangent at an operating point.
  \item The linear model is \hl{local}: gains and even stability can change with the operating point.
  \item Jumps (friction at $v=0$, backlash) need compensation or simulation instead.
\end{itemize}
\end{keybox}
\column{.42\textwidth}
\begin{defbox}[Key terms]
\footnotesize
\term{Operating point}: steady condition we linearise around.\\
\term{Deviation} $\delta x$: small change from that point.\\
\term{Saturation}: output stops growing at a limit.\\
\term{Backlash}: play/gap in a transmission.\\
\term{Stick--slip}: jerky motion caused by friction.
\end{defbox}
\end{columns}

\begin{defbox}[Refresher: complex numbers]
\footnotesize
$z=a+jb$ (engineers write $j=\sqrt{-1}$). Polar form $z=|z|\,e^{j\varphi}$:
$|z|=\sqrt{a^2+b^2}$, $\varphi=\angle z=\operatorname{atan2}(b,a)$.
\end{defbox}
\begin{eqbox}
\[
e^{j\varphi}=\cos\varphi+j\sin\varphi\qquad\text{(Euler)}
\]
\end{eqbox}
\footnotesize
\begin{itemize}
  \item \hl{Multiply}: magnitudes multiply, angles add. \hl{Divide}: magnitudes divide, angles subtract.
  \item Example: $\dfrac{1}{1+j}$: $|1+j|=1.414$, $\angle=45^\circ$ $\Rightarrow$ $0.707\angle-45^\circ$.
  \item For $s=\sigma+j\omega$: $e^{st}=e^{\sigma t}(\cos\omega t+j\sin\omega t)$ $\Rightarrow$ $\sigma$ sets growth/decay, $\omega$ sets oscillation.
\end{itemize}
\end{columns}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Refresher: the Laplace transform as a tool}
\begin{columns}[T]
\column{.46\textwidth}
\begin{defbox}[Refresher: Laplace transform]
\footnotesize
A ``translation'' $f(t)\to F(s)$ that turns \hl{differential equations into algebra} -- like logarithms turn multiplication into addition. We only use the table.
\end{defbox}
\footnotesize
\centering
\begin{tabular}{@{}ll@{}}
\toprule
Time domain $f(t)$ & Laplace $F(s)$\\
\midrule
unit impulse $\delta(t)$ & $1$\\
unit step $1(t)$ & $1/s$\\
ramp $t$ & $1/s^2$\\
$e^{-at}$ & $1/(s+a)$\\
$\sin\omega t$ & $\omega/(s^2+\omega^2)$\\
derivative $\dot f$ & $sF(s)-f(0)$\\
integral $\int_0^tf\,d\tau$ & $F(s)/s$\\
delay $f(t-T)$ & $e^{-sT}F(s)$\\
\bottomrule
\end{tabular}
\column{.52\textwidth}
\small
\textbf{Worked example: RC circuit}, $RC=1$\,ms, input $u$, capacitor voltage $v$ (start at 0\,V):
\begin{enumerate}
  \item ODE: $RC\,\dot v+v=u$
  \item Transform ($\dot v\to sV$): $RC\,s\,V+V=U$
  \item Solve for the ratio:
\end{enumerate}
\begin{eqbox}
\[
\frac{V(s)}{U(s)}=\frac{1}{RC\,s+1}=\frac{1}{0.001\,s+1}
\]
\end{eqbox}
\begin{enumerate}\setcounter{enumi}{3}
  \item 1\,V step: $V=\frac{1}{s(0.001s+1)}$ $\Rightarrow$ table: $v(t)=1-e^{-t/1\,\mathrm{ms}}$\,V -- the familiar RC charging curve.
\end{enumerate}
\end{columns}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Transfer function, poles and zeros}
\begin{columns}[T]
\column{.52\textwidth}
\small
The \term{transfer function} is the ``recipe'' of a system: output over input, with zero initial conditions.
\begin{eqbox}
\[
G(s)=\frac{Y(s)}{U(s)}=\frac{\text{numerator}(s)}{\text{denominator}(s)}
\]
\end{eqbox}
\footnotesize
\begin{itemize}
  \item \term{Poles}: roots of the denominator $\Rightarrow$ decide \hl{stability and speed}.
  \item \term{Zeros}: roots of the numerator $\Rightarrow$ shape the response.
  \item \term{Order} = number of poles = number of energy stores (inductors, masses, \ldots).
  \item \term{DC gain} $G(0)$: output/input after everything has settled.
\end{itemize}
\textbf{Example:} $G(s)=\dfrac{s+2}{(s+1)(s+3)}$: zero $-2$; poles $-1,-3$; $G(0)=2/3$.\\[2pt]
\further{RHP zeros (undershoot), delay $e^{-sT}$ is not a polynomial ratio (Pad\'e approximation).}
\column{.45\textwidth}
\centering
\begin{tikzpicture}[scale=.85]
  \fill[mkCream] (0,-2.2) rectangle (2.2,2.2);
  \node[font=\tiny,mkOrange,anchor=north east] at (2.2,2.2) {unstable (RHP)};
  \fill[mkPale] (-4,-2.2) rectangle (0,2.2);
  \node[font=\tiny,mkBlue,anchor=north west] at (-4,2.2) {stable (LHP)};
  \draw[-{Latex},mkGray] (-4,0)--(2.4,0) node[right]{\scriptsize Re};
  \draw[-{Latex},mkGray] (0,-2.3)--(0,2.4) node[above]{\scriptsize Im};
  \foreach \p in {(-1,1.5),(-1,-1.5),(-3,0)}{\draw[mkBlue,very thick] \p++(-.12,-.12)--++(.24,.24) \p++(-.12,.12)--++(.24,-.24);}
  \draw[mkOrange,very thick] (-2,0) circle (.12);
  \draw[mkOrange,very thick] (1,0) circle (.12);
  \node[font=\tiny,above] at (-1,1.6) {pole $\times$};
  \node[font=\tiny,below] at (-2,-0.12) {zero $\circ$};
  \node[font=\tiny,below,align=center] at (1,-.12) {RHP zero};

\footnotesize Left half-plane (LHP) = stable side.

\begin{frame}{Refresher: matrices for state space}
\begin{columns}[T]
\column{.5\textwidth}
\begin{defbox}[Refresher: matrix times vector]
\footnotesize
Each row of $A$ times the vector $\bm x$ gives one entry:
\[
\begin{bmatrix}0&1\\-4&-2\end{bmatrix}\begin{bmatrix}1\\2\end{bmatrix}
=\begin{bmatrix}0\cdot1+1\cdot2\\-4\cdot1-2\cdot2\end{bmatrix}=\begin{bmatrix}2\\-8\end{bmatrix}
\]
\end{defbox}
\small
Why? Two coupled ODEs
\[
\dot x_1=x_2,\qquad \dot x_2=-4x_1-2x_2
\]
fit in one line: $\dot{\bm x}=A\bm x$ with the matrix above.
\column{.48\textwidth}
\small
\textbf{Eigenvalues} = the natural ``speeds'' of $A$:
\begin{eqbox}
\[
\det(\lambda I-A)=0
\]
\end{eqbox}
\footnotesize
\begin{enumerate}
  \item $\lambda I-A=\begin{bmatrix}\lambda&-1\\4&\lambda+2\end{bmatrix}$
  \item $\det=\lambda(\lambda+2)+4=\lambda^2+2\lambda+4$
  \item Roots: $\lambda=-1\pm j1.73$
\end{enumerate}
\hl{Eigenvalues of $A$ = poles of the system.} This is a mass--spring--damper with $m=1$, $k=4$, $c=2$ ($\omega_n=2$, $\zeta=0.5$).
\end{columns}
\end{frame}

\begin{frame}{State-space representation}
\begin{columns}[T]
\column{.48\textwidth}
\small
\term{State} $\bm x$: the few numbers that describe ``where the system is now'' (positions, speeds, currents, capacitor voltages).
\begin{eqbox}
\[
\dot{\bm x}=A\bm x+B u,\qquad y=C\bm x+D u

\end{eqbox}
\footnotesize
\textbf{Mass--spring--damper}, $x_1$ = position, $x_2$ = velocity, $u=F$:

A=\begin{bmatrix}0&1\\-\tfrac km&-\tfrac cm\end{bmatrix},\;

\[C=\begin{bmatrix}1&0\end{bmatrix},\;D=0\]
\begin{itemize}
  \item Handles many inputs/outputs and initial conditions; basis for observers and state feedback (Day 3).
\end{itemize}
\further{$G(s)=C(sI-A)^{-1}B+D$; the state choice is not unique; canonical forms, \texttt{tf2ss}/\texttt{ss2tf}.}

\begin{frame}{The DC motor in both representations}
\begin{columns}[T]
\column{.44\textwidth}
\centering
\begin{circuitikz}[scale=.78,transform shape,font=\small]
  \draw (0,0) to[V,v<=$v_a$] (0,2.6)
        to[R=$R$,i>^=$i_a$] (2.6,2.6)
        to[L=$L_a$] (4.6,2.6)
        to[short] (4.6,2.2);
  \draw (4.6,0.4) to[short] (4.6,0) -- (0,0);
  \draw[thick,fill=mkPale] (4.6,1.3) circle (.75);
  \node at (4.6,1.3) {M};
  \node[anchor=west] at (5.4,2.0) {$e=k_e\omega$};
  \node[font=\small] at (4.6,2.38) {$+$};
  \node[font=\small] at (4.6,0.22) {$-$};
  \draw[very thick,mkBlue] (5.35,1.3)--(6.6,1.3);
  \draw[thick,fill=mkCream] (6.6,0.7) rectangle (7.0,1.9);
  \node[anchor=north,align=center,font=\footnotesize] at (6.8,0.6) {$J,B$};
  \draw[-{Latex},mkOrange,thick] (5.8,1.75) arc (120:60:.5) node[right]{$\omega$};
\end{circuitikz}\\[2mm]
\footnotesize
\begin{align*}
L_a\tfrac{di_a}{dt}&=v_a-Ri_a-k_e\,\omega\quad\text{(KVL)}\\
J\tfrac{d\omega}{dt}&=k_t\,i_a-B\omega\quad\text{(Newton)}
\end{align*}
SI units: $k_t$ [Nm/A] $=k_e$ [V\,s/rad].
\column{.54\textwidth}
\footnotesize
\textbf{State space}: two energy stores $\Rightarrow$ two states $\bm x=[i_a,\ \omega]^{\!\top}$, $u=v_a$, $y=\omega$:
\[
A=\begin{bmatrix}-\frac{R}{L_a}&-\frac{k_e}{L_a}\\[2pt]\frac{k_t}{J}&-\frac{B}{J}\end{bmatrix},\;
B=\begin{bmatrix}\frac1{L_a}\\[2pt]0\end{bmatrix},\;
C=\begin{bmatrix}0&1\end{bmatrix}
\]
\textbf{Transfer function} (Laplace of both equations, eliminate $I_a$):
\begin{eqbox}
\[
\frac{\Omega(s)}{V_a(s)}=\frac{k_t}{(L_as+R)(Js+B)+k_tk_e}
\]
\end{eqbox}
\begin{itemize}
  \item The $k_tk_e$ term is \term{back-EMF}: the motor has a \hl{built-in feedback loop} (faster $\Rightarrow$ less current).
  \item Load torque $T_L$ is a second input (a disturbance).
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Session 2.2 -- Summary and key terms}
\begin{columns}[T]
\column{.55\textwidth}
\begin{keybox}[Key take-aways]
\footnotesize
\begin{itemize}
  \item Laplace turns ODEs into algebra: $G(s)=Y(s)/U(s)$.
  \item \hl{Poles} decide stability and speed; left half-plane = stable.
  \item First order ($K,\tau$) and second order ($K,\zeta,\omega_n$) describe most servo sub-systems.
  \item State space $\dot{\bm x}=A\bm x+Bu$: eigenvalues of $A$ = poles.
  \item DC motor: back-EMF is built-in feedback; feedback makes gains insensitive to plant changes.
\end{itemize}
\end{keybox}
\column{.42\textwidth}
\begin{defbox}[Key terms]
\footnotesize
\term{Transfer function} $G(s)$: output/input in Laplace domain.\\
\term{Pole / zero}: root of denominator / numerator.\\
\term{Time constant} $\tau$: time to reach 63\,\%.\\
\term{Damping ratio} $\zeta$: how bouncy (0 = no damping).\\
\term{State}: variables that store energy/memory.
\end{defbox}
\end{columns}
\end{frame}

\term{Dead time} $\theta$: like a shower -- turn the tap and the water temperature changes only seconds later.
\begin{eqbox}
\[
G(s)=\frac{K\,e^{-\theta s}}{\tau s+1}
\]
\end{eqbox}
\footnotesize
\textbf{Two-point method}, unit step input ($\Delta u=1$):
\begin{enumerate}
  \item Final change $\Delta y=2$ $\Rightarrow$ $K=\Delta y/\Delta u=2$.
  \item Read $t_{28}=1.00$\,s and $t_{63}=1.86$\,s.
  \item $\tau=1.5\,(t_{63}-t_{28})=1.5\cdot0.86=1.29$\,s.
  \item $\theta=t_{63}-\tau=1.86-1.29=0.57$\,s.
\end{enumerate}
Good for thermal/process loops; too coarse for springy mechanics.

  \addplot[mkOrange,very thick,domain=0:2,samples=120] {0.6*sin(360*x-60)}; \addlegendentry{output $y$}
  \draw[mkNavy,<->] (axis cs:0.25,0)--(axis cs:0.25,1) node[pos=.5,left,font=\tiny]{$\hat u$};
  \draw[mkOrange,<->] (axis cs:1.42,0)--(axis cs:1.42,0.6) node[pos=.5,right,font=\tiny]{$\hat y$};
  \draw[mkGray,<->] (axis cs:1.25,1.05)--(axis cs:1.4167,1.05) node[midway,above,font=\tiny]{lag};
\end{axis}
\end{tikzpicture}\\[1mm]
\footnotesize Shake a ruler slowly: the tip follows. Shake it fast: the tip moves less and \hl{lags behind}.
\column{.5\textwidth}
\begin{defbox}[Refresher: frequency response]
\footnotesize
A linear stable system driven by a sine answers with a sine of the \hl{same frequency}, only scaled and shifted.
\end{defbox}
\begin{eqbox}
\[
y=|G(j\omega)|\,\hat u\,\sin\!\big(\omega t+\angle G(j\omega)\big)
\]
\end{eqbox}
\footnotesize
for input $u=\hat u\sin\omega t$. \textbf{Recipe}: replace $s$ by $j\omega$.\\
Example $G=\dfrac{1}{s+1}$:
\begin{enumerate}
  \item $\omega=1$\,rad/s: $G(j1)=\dfrac{1}{1+j}=0.707\angle-45^\circ$
  \item $\omega=10$\,rad/s: $\dfrac{1}{1+10j}=0.0995\angle-84^\circ$
\end{enumerate}
Faster input $\Rightarrow$ smaller output and more lag.
\end{columns}
\end{frame}

\begin{frame}{How to read a Bode plot}
\begin{columns}[T]
\column{.42\textwidth}
\footnotesize
A \term{Bode plot} = frequency response for all $\omega$: magnitude in dB (top) and phase in degrees (bottom), both over a log frequency axis.\\[1mm]
\footnotesize
\textbf{Reading it}:
\begin{enumerate}
  \item pick $\omega$ (e.g.\ 0.1\,rad/s)
  \item read magnitude: 40\,dB $\Rightarrow$ gain $\times100$
  \item read phase: $\approx-85^\circ$ (output lags)
\end{enumerate}
\textbf{Straight-line rules}:
\begin{itemize}
  \item each pole: $-20$\,dB/decade after its corner, phase drops by $90^\circ$

\small
\term{Loop gain} $L(s)=C(s)G(s)$: go once around the loop. \term{Sensitivity} $S$: what is left of a disturbance; \term{complementary sensitivity} $T$: how well $y$ follows $r$.
\begin{eqbox}
\[
S=\frac{1}{1+L},\qquad T=\frac{L}{1+L}
\]
\end{eqbox}
\footnotesize
\begin{itemize}
  \item Cruise control: $L=9$ $\Rightarrow$ $S=0.1$, $T=0.9$; $S+T=1$ always.
  \item Big $|L|$ $\Rightarrow$ small $S$ $\Rightarrow$ good tracking -- \hl{but} sensor noise also reaches $y$.
\end{itemize}

\begin{frame}{Classical tuning: Ziegler--Nichols (1942)}
\begin{columns}[T]
\column{.46\textwidth}
\small
\textbf{Recipe:} use P only, raise $K_p$ until the loop just keeps oscillating. Note this \term{ultimate gain} $K_u$ and the period $T_u$.
\begin{center}
\footnotesize
\begin{tabular}{@{}lccc@{}}
\toprule
 & $K_p$ & $T_i$ & $T_d$\\
\midrule
P & $0.5K_u$ & -- & --\\
PI & $0.45K_u$ & $T_u/1.2$ & --\\
PID & $0.6K_u$ & $T_u/2$ & $T_u/8$\\
\bottomrule
\end{tabular}
\end{center}
\further{step-response version: PID $K_p=1.2\tau/(K\theta)$, $T_i=2\theta$, $T_d=\theta/2$.}
\column{.52\textwidth}
\begin{defbox}[Worked example]
\footnotesize
Plant $G=1/(s+1)^3$.
\begin{enumerate}
  \item Each pole lags up to $90^\circ$; total $-180^\circ$ when $3\arctan\omega=180^\circ$ $\Rightarrow$ $\omega_u=\sqrt3=1.73$\,rad/s.
  \item There $|G|=1/(1+3)^{3/2}=1/8$ $\Rightarrow$ $K_u=8$.
  \item $T_u=2\pi/1.73=3.63$\,s.
  \item PID: $K_p=4.8$, $T_i=1.81$\,s, $T_d=0.45$\,s.
\end{enumerate}
\end{defbox}
\begin{keybox}[Caution]
\footnotesize
Gives fast but \hl{aggressive} loops (big overshoot). Pushing a real machine to oscillation can be unsafe. Use ZN only as a starting point [\AA str\"om \& H\"agglund, 2006].
\end{keybox}
\end{columns}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Model-based tuning: one knob for speed}
\begin{columns}[T]
\column{.5\textwidth}
\small
For an FOPDT model $K e^{-\theta s}/(\tau s+1)$ and a PI controller, choose the \hl{desired closed-loop time constant} $\tau_c$:
\begin{eqbox}
\[
K_p=\frac{\tau}{K(\tau_c+\theta)},\qquad T_i=\tau
\]
\end{eqbox}
\footnotesize
\begin{itemize}
  \item Smaller $\tau_c$ $\Rightarrow$ faster but less robust.
  \item SIMC default $\tau_c=\theta$: good trade-off [Skogestad, 2003].
  \item Lambda tuning: same formula, $\tau_c=\lambda$, often $\lambda=\tau$ (gentle).
\end{itemize}
\further{SIMC $T_i=\min\{\tau,4(\tau_c+\theta)\}$; Cohen--Coon gives $K_p=1.06$, $T_i=1.0$\,s here (aggressive).}
\column{.48\textwidth}
\begin{defbox}[Example: FOPDT from session 2.3]
\footnotesize
$K=2$, $\tau=1.29$\,s, $\theta=0.57$\,s
\begin{enumerate}
  \item SIMC, $\tau_c=\theta=0.57$\,s: $K_p=\dfrac{1.29}{2\,(0.57+0.57)}=0.57$
  \item Lambda, $\lambda=\tau=1.29$\,s: $K_p=\dfrac{1.29}{2\,(1.29+0.57)}=0.35$
  \item Both: $T_i=\tau=1.29$\,s
  \item ZN step rule (PI): $K_p=0.9\tau/(K\theta)=1.02$
\end{enumerate}
\end{defbox}
\footnotesize
A 3:1 spread in $K_p$: the rules aim at \hl{different robustness}, there is no single ``right'' answer.
\end{columns}
\end{frame}

\begin{frame}{Derivative action: always filter it}
\begin{columns}[T]
\column{.5\textwidth}
\centering
\begin{tikzpicture}
\begin{axis}[mkplot,width=7.0cm,height=5.0cm,xmode=log,xmin=0.01,xmax=1000,ymin=-30,ymax=50,
  xlabel={$\omega T_d$},ylabel={gain of D-part [dB]},legend pos=north west]
  \addplot[mkRed,thick,dashed,domain=-2:3,samples=100] ({10^x},{20*x}); \addlegendentry{ideal $K_pT_ds$}
  \addplot[mkBlue,very thick,domain=-2:3,samples=150] ({10^x},{20*x-10*log10(1+(10^x/10)^2)}); \addlegendentry{filtered, $N=10$}
  \addplot[mkOrange,thick,domain=-2:3,samples=150] ({10^x},{20*x-10*log10(1+(10^x/5)^2)}); \addlegendentry{filtered, $N=5$}
  \node[font=\tiny,anchor=west] at (axis cs:15,26) {limit $20\log N$};
\end{axis}
\end{tikzpicture}
\column{.48\textwidth}
\small
A pure derivative amplifies fast wiggles without limit: sensor noise becomes huge.
\begin{eqbox}
\[
D(s)=\frac{K_pT_d\,s}{1+T_ds/N},\qquad N\approx5\ldots20
\]
\end{eqbox}
\footnotesize
\begin{itemize}
  \item The filter caps the high-frequency D-gain at $K_pN$.
  \item \hl{Derivative on measurement} ($-\dot y$ instead of $\dot e$): no ``kick'' when the set-point jumps.
  \item Servo practice: D on position = velocity feedback; in a cascade it is the inner speed loop.
  \item Example: 20-bit encoder, 125\,$\mu$s sampling: one count gives a speed jump of 0.048\,rad/s.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Set-point weighting: gentle on commands, firm on disturbances}
\begin{columns}[T]
\column{.5\textwidth}
\centering
\begin{tikzpicture}

  \addplot[mkOrange,thick,dashed] table[x=t,y=b05] {data/spweight.dat}; \addlegendentry{$b=0.5$}
  \addplot[mkBlue,very thick] table[x=t,y=b0] {data/spweight.dat}; \addlegendentry{$b=0$}
\end{axis}
\end{tikzpicture}\\
\footnotesize PI, $K_p=2$, $T_i=2.5$\,s, $G=1/(s+1)^3$. Overshoot: 33\,\%, 3.4\,\%, 0.3\,\%.
\column{.48\textwidth}
\small
Use only part ($b$) of the set-point in the P-term:
\begin{eqbox}
\[
u=K_p\Big[(b\,r-y)+\frac1{T_i}\!\int(r-y)\,dt\Big]
\]
\end{eqbox}
\footnotesize
\begin{itemize}
  \item The I-term keeps the \hl{true} error, so the final value is still exact.
  \item Disturbance response and stability are \hl{unchanged}: $b$ only shapes the reaction to new set-points.
  \item Recipe: tune $K_p$, $T_i$ for disturbances first, then pick $b$.
  \item Servo equivalent: smooth trajectory + feedforward (next slides).
\end{itemize}
\end{columns}
\end{frame}

\footnotesize Plant $1/(s+1)$, PI $K_p=4$, $T_i=0.8$\,s, limit $|u|\le1.2$.
\column{.42\textwidth}
\small
Cruise control uphill: throttle is at 100\,\% but the car is still too slow.
\footnotesize
\begin{itemize}
  \item The integrator keeps adding up error although the throttle \hl{cannot} open further: it ``winds up''.
  \item On the hill-top it must first ``unwind'' $\Rightarrow$ big overshoot (18\,\% here) and slow settling.
\end{itemize}
\begin{defbox}[Anti-windup]
\footnotesize
\textbf{Clamping}: stop integrating while the output is at its limit and the error pushes further. Simple, standard in drives.\\[2pt]
\textbf{Back-calculation}: feed $(u-v)/T_t$ back into the integrator so it tracks the limit ($T_t\approx T_i/2$).
\end{defbox}
\further{in cascades, pass the inner loop's limit to the outer integrator.}
\end{columns}
\end{frame}

\begin{frame}{Session 2.4 -- Summary and key terms}
\begin{columns}[T]
\column{.55\textwidth}
\begin{keybox}[Key take-aways]
\footnotesize
\begin{itemize}
  \item Feedback = measure, compare, correct; it fights unknown disturbances.
  \item P reacts to now, I removes the steady error, D damps (filter it!).
  \item Tune with a model ($\tau_c$ knob) rather than by pushing to instability.
  \item Every saturating integrator needs \hl{anti-windup}.
  \item Feedforward and set-point weighting improve tracking without changing stability.
  \item Servo drives: current $\to$ speed $\to$ position cascade, tuned inside-out.
\end{itemize}
\end{keybox}
\column{.42\textwidth}
\begin{defbox}[Key terms]
\footnotesize
\term{Loop gain} $L=CG$: gain once around the loop.\\
\term{Sensitivity} $S=1/(1+L)$: fraction of disturbance left.\\
\term{Windup}: integrator grows while the actuator is at its limit.\\
\term{Feedforward}: act on the known command before an error appears.\\
\term{Cascade}: loops nested inside each other.
\end{defbox}
\end{columns}

  \end{scope}
\end{tikzpicture}\\[3mm]
\footnotesize\raggedright
Shower analogy: the hot water arrives late (dead time). If you over-react to what you feel now, the temperature swings ever more -- \hl{feedback can create instability}.
\column{.48\textwidth}
\small
Closed-loop poles are the roots of the \term{characteristic equation} $1+L(s)=0$.
\begin{eqbox}
\[
\text{stable}\iff\text{all closed-loop poles have }\operatorname{Re}(p)<0
\]
\end{eqbox}
\footnotesize
\textbf{Example} $L(s)=\dfrac{K}{s(s+2)(s+5)}$:
\begin{enumerate}
  \item $1+L=0$ $\Rightarrow$ $s^3+7s^2+10s+K=0$
  \item $K=20$: poles $-5.88$ and $-0.56\pm j1.76$ $\Rightarrow$ stable
  \item $K=100$: two poles in the right half-plane $\Rightarrow$ unstable
\end{enumerate}
Too much gain destabilises -- we will see why on the next slides.
\end{columns}
\end{frame}

\begin{frame}{Steady-state error and system type}
\begin{columns}[T]
\column{.48\textwidth}
\small
Final value theorem (only for a \hl{stable} loop):
\begin{eqbox}
\[
e_\infty=\lim_{t\to\infty}e(t)=\lim_{s\to0}s\,E(s),\quad E=\frac{R}{1+L}
\]
\end{eqbox}
\footnotesize
\term{System type} = number of integrators ($1/s$) in $L(s)$.
\begin{center}
\begin{tabular}{@{}lccc@{}}
\toprule
Type & step & ramp & example\\
\midrule
0 & $\frac{1}{1+L(0)}$ & $\infty$ & cruise control, P only\\[3pt]
1 & 0 & $\frac{1}{K_v}$ & PI, or position loop\\[3pt]
2 & 0 & 0 & position loop + I\\
\bottomrule
\end{tabular}
\end{center}
$K_v=\lim_{s\to0}sL(s)$ [s$^{-1}$], the velocity constant.
\column{.5\textwidth}
\begin{defbox}[Servo example: following error]
\footnotesize
\begin{enumerate}
  \item Position P gain $K_{pp}=126$\,s$^{-1}$ around a fast speed loop: $L\approx K_{pp}/s$ $\Rightarrow$ type 1, $K_v=126$\,s$^{-1}$.
  \item Table moves at 0.2\,m/s, 10\,mm ball-screw pitch $\Rightarrow$ motor speed $\dot\theta=126$\,rad/s (a ramp in $\theta$).
  \item $e_\infty=\dot\theta/K_v=126/126=1$\,rad.
  \item $1$\,rad $\times10\,\mathrm{mm}/2\pi=1.6$\,mm lag behind the command.
\end{enumerate}
\end{defbox}
\footnotesize
Cure: velocity feedforward (previous session) removes this lag.
\end{columns}
\end{frame}

\begin{frame}{Going further: Routh--Hurwitz and Nyquist (concepts only)}
\begin{columns}[T]
\column{.45\textwidth}
\begin{defbox}[Routh--Hurwitz]
\footnotesize
Decides stability from the polynomial coefficients without solving for the roots. For $s^3+7s^2+10s+K$:
\begin{center}
\begin{tabular}{@{}c|cc@{}}
$s^3$ & 1 & 10\\
$s^2$ & 7 & $K$\\
$s^1$ & $(70-K)/7$ & 0\\
$s^0$ & $K$ & \\
\end{tabular}
\end{center}
No sign change in the first column $\iff$ $0<K<70$: the same limit as the root locus.
\end{defbox}
\column{.52\textwidth}
\begin{columns}[T]
\column{.52\linewidth}
\centering
\begin{tikzpicture}

\end{axis}
\end{tikzpicture}
\column{.46\linewidth}
\footnotesize
\textbf{Nyquist}: plot $L(j\omega)$ in the complex plane. For a stable open loop, the closed loop is stable if the curve does \hl{not encircle $-1$}.

\column{.43\textwidth}
\small
The loop oscillates if a signal comes back around the loop with gain 1 and $-180^\circ$. Margins = \hl{safety distance} from that point.
\footnotesize
\begin{enumerate}
  \item Find $\omega_c$ where $|L|=0$\,dB (1.52\,rad/s).
  \item Read the phase there: $-144^\circ$ $\Rightarrow$ \term{phase margin} PM $=180^\circ-144^\circ=36^\circ$.
  \item Find where the phase is $-180^\circ$ (3.16\,rad/s); $|L|=-10.9$\,dB there $\Rightarrow$ \term{gain margin} GM $=10.9$\,dB ($\times3.5$).
\end{enumerate}
\begin{itemize}
  \item Targets: PM $45$--$60^\circ$, GM $\ge6$\,dB.
  \item A delay $T$ eats phase $\omega_cT$: here $0.63\,\mathrm{rad}/1.52=0.41$\,s delay would make it oscillate.
\end{itemize}
\end{columns}

\begin{frame}{Sensitivity, robustness and loop shaping}
\begin{columns}[T]
\column{.5\textwidth}
\centering
\begin{tikzpicture}
\begin{axis}[mkplot,width=7.0cm,height=4.2cm,xmode=log,xmin=0.01,xmax=100,ymin=-40,ymax=15,

  \addplot[mkOrange,very thick] table[x=w,y=S] {data/loop_bode.dat}; \addlegendentry{$|S|$}
  \addplot[mkBlue,very thick] table[x=w,y=T] {data/loop_bode.dat}; \addlegendentry{$|T|$}
  \addplot[mkGray,dashed,domain=0.01:100] {0};
  \node[font=\tiny,mkOrange,anchor=south west] at (axis cs:2.2,6.6) {$M_s=2.12$ (6.5\,dB)};
\end{axis}
\end{tikzpicture}
\begin{tikzpicture}
\begin{axis}[mkplot,width=7.0cm,height=3.5cm,xmin=0,xmax=12,ymin=0,ymax=1.9,xlabel={time [s]},ylabel={$y$},
  legend pos=north east,legend columns=3]
  \addplot[mkSky,thick] table[x=t,y=k10] {data/loop_step.dat}; \addlegendentry{$K=10$}
  \addplot[mkBlue,very thick] table[x=t,y=k20] {data/loop_step.dat}; \addlegendentry{$20$}
  \addplot[mkRed,thick] table[x=t,y=k50] {data/loop_step.dat}; \addlegendentry{$50$}
\end{axis}
\end{tikzpicture}
\column{.48\textwidth}
\small
$|S(j\omega)|$ = how much of a disturbance at frequency $\omega$ survives the loop.
\begin{eqbox}
\[
M_s=\max_\omega|S(j\omega)|\quad(\text{aim: }1.2\ldots2)
\]
\end{eqbox}
\footnotesize
\begin{itemize}
  \item Low frequency: $|S|\ll1$ $\Rightarrow$ disturbances removed.
  \item Near crossover: $|S|>1$ $\Rightarrow$ feedback \hl{amplifies} disturbances. Peak $M_s$ = 1/(closest distance to $-1$).
  \item $K=20$: $M_s=2.1$, $36^\circ$ PM $\Rightarrow$ ringing; $K=10$ is calmer.
  \item \term{Loop shaping}: make $|L|$ big at low frequency (tracking), small at high frequency (noise, resonances).
\end{itemize}
\further{$\mathrm{GM}\ge M_s/(M_s-1)$, $\mathrm{PM}\ge2\arcsin(1/2M_s)$; waterbed effect $\int_0^\infty\ln|S|\,d\omega=0$.}
\end{columns}
\end{frame}

\begin{frame}{Session 2.5 -- Summary, design targets and key terms}
\begin{columns}[T]
\column{.36\textwidth}
\begin{keybox}[Key take-aways]
\footnotesize
\begin{itemize}
  \item Stable $\iff$ closed-loop poles in the left half-plane.
  \item $\zeta$ sets overshoot; PM $\approx100\zeta$.
  \item System type sets the steady-state error.
  \item More gain $\Rightarrow$ poles move right, margins shrink.
\end{itemize}

$\omega_c$ vs.\ resonance & $<\omega_r/3$\\
\bottomrule
\end{tabular}\\[3pt]

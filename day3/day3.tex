\begin{frame}{Refresher: what does a filter do?}
\begin{columns}[T]
\begin{column}{.48\textwidth}
\begin{defbox}[Refresher: a filter is a music equaliser]
A \term{filter} passes some frequencies and weakens others.
\begin{itemize}\setlength\itemsep{0mm}
\item \term{low-pass}: treble down, bass kept
\item \term{high-pass}: bass down (removes DC)
\item \term{notch}: removes one tone (50\,Hz hum)
\end{itemize}
\end{defbox}
\small
\vspace{-1mm}
\begin{eqbox}
\[ |F(f)|=\frac{1}{\sqrt{1+(f/f_f)^2}},\qquad f_f=\frac{1}{2\pi RC}\ \text{[Hz]} \]
\end{eqbox}
RC low-pass: $R=\SI{1}{k\ohm}$, $C=\SI{1}{\micro F}$ $\Rightarrow$ $f_f=\SI{159}{Hz}$; at \SI{1.59}{kHz} only $0.1$ passes.
\end{column}
\begin{column}{.50\textwidth}
\centering
\begin{tikzpicture}
\begin{axis}[mkplot,xmode=log,width=7.2cm,height=4.3cm,xmin=0.01,xmax=100,ymin=-45,ymax=5,
  xlabel={frequency $f/f_f$},ylabel={gain [dB]},legend style={at={(0.03,0.03)},anchor=south west}]
\addplot[mkBlue,thick,domain=-2:2,samples=120] ({10^x},{-10*log10(1+10^(2*x))});\addlegendentry{low-pass}
\addplot[mkOrange,thick,domain=-2:2,samples=120] ({10^x},{20*x-10*log10(1+10^(2*x))});\addlegendentry{high-pass}
\addplot[mkMid,thick,densely dashed,domain=-2:2,samples=200] ({10^x},{10*log10(((1-10^(2*x))^2+(0.04*10^x)^2)/((1-10^(2*x))^2+(10^x)^2))});\addlegendentry{notch}
\end{axis}
\end{tikzpicture}

\begin{keybox}[Take-away]
Every filter that weakens a signal also \hl{delays} it a little. In a control loop that delay is a cost.
\end{keybox}
\end{column}
\end{columns}
\end{frame}

\textbf{What we filter against}
\begin{itemize}\setlength\itemsep{.3mm}

\begin{frame}{Low-pass filter: cut-off, order and phase}
\begin{columns}[T]
\begin{column}{.43\textwidth}
\small
\begin{eqbox}
\[ F_1(s)=\frac{\omega_f}{s+\omega_f} \]
\end{eqbox}
\begin{itemize}\setlength\itemsep{.4mm}
\item \term{cut-off} $\omega_f$ [rad/s]: the $-3$\,dB point
\item \term{order} $n$: steeper roll-off, $-20n$\,dB per decade
\item \term{phase}: $-45^\circ$ per order at $\omega_f$
\item 2nd order with low damping \hl{peaks}: it amplifies near $\omega_f$
\end{itemize}
\textbf{Take-away:} higher order = cleaner signal but much more phase lag.
\end{column}

\end{axis}
\end{tikzpicture}
\end{column}
\begin{column}{.51\textwidth}
\small
\begin{eqbox}
\[ \Delta\varphi=\arctan\frac{\omega_c}{\omega_f}\quad\text{(1st-order LPF)} \]
\end{eqbox}
\textbf{Worked example.} Velocity loop crossover $f_c=150$\,Hz; we allow $10^\circ$ loss.
\begin{enumerate}\setlength\itemsep{.3mm}
\item need $f_c/f_f\le\tan10^\circ=0.176$
\item $f_f\ge 150\,\text{Hz}/0.176=\SI{850}{Hz}$
\item check: $\arctan(150/850)=10.0^\circ$ \checkmark
\end{enumerate}
\end{column}
\end{columns}
\vspace{1mm}
\begin{keybox}[Trade-off]
Lower $\omega_f$ = quieter motor but less phase margin. Rule of thumb: $\omega_f\approx(5\ldots10)\,\omega_c$.
\end{keybox}
\end{frame}

\begin{frame}{High-pass filter: removing offsets}
\begin{columns}[T]
\begin{column}{.48\textwidth}
\small
\begin{eqbox}
\[ F_{HP}(s)=\frac{s}{s+\omega_h} \]
\end{eqbox}
\begin{itemize}\setlength\itemsep{.4mm}
\item blocks DC: removes sensor \term{offsets} and slow drift
\item \term{washout}: passes only \emph{changes} (vibration, acceleration)
\item gives phase \hl{lead} (up to $+90^\circ$) below $\omega_h$
\end{itemize}
\begin{notebox}
\textbf{Pitfall:} never in the \emph{main} feedback path --- the loop would no longer see a constant position error and could not hold a load.
\end{notebox}
\end{column}

\begin{frame}{Notch filter: cutting out one frequency}
\begin{columns}[T]
\begin{column}{.45\textwidth}
\small
\textbf{Picture:} a playground swing has one natural frequency; a mechanical axis too. A \term{notch} removes the controller's ``push'' at exactly that frequency.
\begin{eqbox}
\[ N(s)=\frac{s^2+2\zeta_z\omega_n s+\omega_n^2}{s^2+2\zeta_p\omega_n s+\omega_n^2} \]

\begin{itemize}\setlength\itemsep{.3mm}
\item \term{centre} $\omega_n$: set to the resonance
\item \term{depth}: $\zeta_z/\zeta_p$ (e.g.\ $0.1\to-20$\,dB)
\item \term{width}: set by $\zeta_p$ --- wide = robust to drift, but more phase lag
\end{itemize}

\begin{frame}{Phase-lead compensator: buying phase back}
\begin{columns}[T]
\begin{column}{.46\textwidth}
\small
\begin{eqbox}
\[ C_{\text{lead}}(s)=K\,\frac{1+\alpha T s}{1+T s},\qquad \alpha>1 \]
\end{eqbox}
\begin{itemize}\setlength\itemsep{.4mm}
\item a zero before a pole: phase goes \hl{up} between them
\item maximum lead at $\omega_m=1/(T\sqrt\alpha)$ --- put it at $\omega_c$
\item $\alpha=3\to30^\circ$, $\alpha=7.5\to50^\circ$, $\alpha=20\to65^\circ$
\item price: high-frequency gain $\times\alpha$ $\Rightarrow$ more noise
\end{itemize}
\textbf{Take-away:} a lead is a PD controller with a filtered D.
\end{column}

\begin{frame}{Phase-lag compensator and lead--lag}
\begin{columns}[T]
\begin{column}{.47\textwidth}
\small
\begin{eqbox}
\[ C_{\text{lag}}(s)=\beta\,\frac{1+T s}{1+\beta T s},\qquad \beta>1 \]
\end{eqbox}
\begin{itemize}\setlength\itemsep{.4mm}
\item boosts gain at \hl{low} frequency by $\beta$ $\Rightarrow$ smaller steady error, stiffer against load
\item leaves high frequencies (and noise) alone
\item put its zero a decade below $\omega_c$: costs only $\approx5^\circ$
\item \term{lead--lag}: lead buys PM, lag buys stiffness
\end{itemize}
\begin{notebox}

\begin{frame}{Refresher: matrices and eigenvalues}
\begin{columns}[T]
\begin{column}{.52\textwidth}
\begin{defbox}[Refresher: eigenvalues = natural rhythms]
A matrix $A$ maps a vector to a new vector. Special directions are only \emph{stretched}:
\[ A\bm v=\lambda\bm v \]
$\lambda$ = \term{eigenvalue}. Find them from $\det(\lambda I-A)=0$.
\end{defbox}
\small
\textbf{Example:} $A=\begin{bmatrix}0&1\\-2&-3\end{bmatrix}$
\begin{enumerate}\setlength\itemsep{.3mm}
\item $\det(\lambda I-A)=\lambda(\lambda+3)+2=\lambda^2+3\lambda+2$
\item roots: $\lambda_1=-1$, $\lambda_2=-2$ [1/s]
\item so $\dot{\bm x}=A\bm x$ contains $e^{-t}$ and $e^{-2t}$: both decay
\end{enumerate}
\end{column}
\begin{column}{.46\textwidth}
\centering
\begin{tikzpicture}[x=1cm,y=1cm]
  \fill[mkPale] (-2.6,-1.6) rectangle (0,1.6);

  \node[font=\tiny,below] at (-1,-0.1) {$-1$};
  \node[font=\tiny,below] at (-2,-0.1) {$-2$};

\end{tikzpicture}

\vspace{2mm}
\begin{keybox}[Take-away]
Eigenvalues of $A$ = poles of the system. Like a guitar string's natural notes, they set how the system ``rings'' and whether it dies out (left half-plane).
\end{keybox}
\end{column}

\begin{frame}{State-space model of a servo axis}
\begin{columns}[T]
\begin{column}{.5\textwidth}
\small
\term{State} $\bm x$: the smallest set of numbers that remembers everything about the system (here: where it is, how fast it moves).
\begin{eqbox}
\[ \dot{\bm x}=A\bm x+B u,\qquad y=C\bm x \]
\end{eqbox}
\textbf{Axis} $J\dot\omega=-B_v\omega+u$ with $\bm x=[\theta\ \ \omega]^{\mathsf T}$, $u=$ torque:
\[
A=\begin{bmatrix}0&1\\0&-a\end{bmatrix},\;
B=\begin{bmatrix}0\\ b\end{bmatrix},\;
C=\begin{bmatrix}1&0\end{bmatrix}
\]
$a=B_v/J$, $b=1/J$.
\end{column}
\begin{column}{.48\textwidth}
\small
\textbf{Numbers for this session:}\\
$J=\SI{1e-3}{kg\,m^2}$, $B_v=\SI{2e-3}{N\,m\,s/rad}$\\
$\Rightarrow a=\SI{2}{s^{-1}}$, $b=\SI{1000}{1/(kg\,m^2)}$.
\vspace{2mm}
\begin{itemize}\setlength\itemsep{.4mm}
\item row 1: $\dot\theta=\omega$ (definition of speed)
\item row 2: Newton for rotation
\item one state per energy store (mass, inertia, inductor, \dots)
\item same model works for many inputs and outputs
\end{itemize}
\begin{notebox}
Adding the motor current $i$ as a third state gives a $3\times3$ model --- same idea, one more row.
\end{notebox}
\end{column}
\end{columns}
\end{frame}

\begin{column}{.48\textwidth}
\small
\begin{eqbox}
\[ u=-K\bm x+N r \]\[ \Rightarrow\ \dot{\bm x}=(A-BK)\bm x+BN\,r \]
\end{eqbox}
\begin{itemize}\setlength\itemsep{.4mm}
\item new system matrix $A-BK$: its eigenvalues are the \hl{closed-loop poles}
\item choose $K$ $\Rightarrow$ choose the poles (speed and damping)
\item $N$ scales $r$ so that $y\to r$
\item needs every state --- measured or \term{estimated}
\end{itemize}
\textbf{Limit:} faster poles need more torque and amplify noise.
\end{column}
\end{columns}

\begin{frame}{Worked example: placing the poles of the servo axis}
\begin{columns}[T]
\begin{column}{.56\textwidth}
\small
\textbf{Goal:} $\omega_n=\SI{100}{rad/s}$, $\zeta=0.7$. With $K=[k_1\ k_2]$:
\begin{enumerate}\setlength\itemsep{.6mm}
\item desired polynomial: $s^2+2\zeta\omega_n s+\omega_n^2=s^2+140s+10^4$
\item closed loop: $\det(sI-A+BK)=s^2+(a+bk_2)s+bk_1$
\item match the last terms: $1000\,k_1=10^4\Rightarrow k_1=\SI{10}{N\,m/rad}$
\item match the middle terms: $2+1000\,k_2=140\Rightarrow k_2=\SI{0.138}{N\,m\,s/rad}$
\end{enumerate}

\[ K=[\,10\ \ \ 0.138\,] \]
\end{eqbox}

\begin{frame}{The observer: estimating states we do not measure}
\begin{columns}[T]
\begin{column}{.47\textwidth}
\small
\textbf{Picture:} a ship's navigator without GPS uses a model (speed $\times$ time) and corrects it whenever a lighthouse is seen.
\begin{eqbox}
\[ \dot{\hat{\bm x}}=\underbrace{A\hat{\bm x}+Bu}_{\text{model}}+\underbrace{L\,(y-C\hat{\bm x})}_{\text{correction}} \]
\end{eqbox}
\begin{itemize}\setlength\itemsep{.3mm}
\item estimation error dies out with poles $\mathrm{eig}(A-LC)$
\item make them 2--6$\times$ faster than the controller poles
\item \term{observable}: the measurement ``sees'' every state (encoder angle: yes; speed only: angle cannot be recovered)
\end{itemize}
\end{column}

\begin{frame}{Going further: controllability, Ackermann's formula and LQR}
\gofurther
\begin{columns}[T]
\begin{column}{.48\textwidth}
\begin{defbox}[Controllability and Ackermann]
\small
\term{Controllable}: the input can push every state anywhere. Test: $\mathcal C=[B\ \ AB\ \cdots]$ has full rank.\\[1mm]
\term{Ackermann's formula} computes $K$ directly from $\mathcal C$ and the desired poles --- what software does for larger systems. For our $2\times2$ axis it gives the same $K=[10\ \ 0.138]$.
\end{defbox}
\end{column}
\begin{column}{.50\textwidth}
\begin{defbox}[LQR: optimal state feedback]
\small
Instead of choosing poles, choose what you care about:
\[ J=\int_0^\infty\!\left(\bm x^{\mathsf T}Q\bm x+u^{\mathsf T}Ru\right)dt \]
large $Q$ = ``errors are expensive'', large $R$ = ``torque is expensive''. Software returns the best $K$; good margins come for free. Used for multi-mass axes and crane anti-sway.
\end{defbox}
\end{column}
\end{columns}
\src{[Ogata, 2010, ch.~10]; [{\AA}str\"om \& Murray, 2021, ch.~7]; [Anderson \& Moore, 1990].}

\begin{frame}{Gain scheduling: a pre-computed gain table}

\begin{frame}{Refresher: sampling a signal}
\begin{columns}[T]
\begin{column}{.46\textwidth}
\begin{defbox}[Refresher: sampling = taking snapshots]
A film camera shows motion as 24 photos per second. A digital controller sees a signal only at the instants $t=kT$:
\[ f_s=\frac1T\quad\text{[Hz]},\qquad y_k=y(kT) \]
Between samples the output is \term{held} constant (zero-order hold, ZOH): a staircase.
\end{defbox}
\small
\textbf{Example:} $f_s=8$\,kHz $\Rightarrow T=\SI{125}{\micro s}$; a 100\,Hz sine gets 80 samples per period.

\begin{frame}{From the $s$-plane to the $z$-plane}
\begin{columns}[T]
\begin{column}{.46\textwidth}
\small
In discrete time a pole is a number $z$ that multiplies the signal every sample: $x_{k+1}=z\,x_k$.
\begin{eqbox}
\[ z=e^{sT}\qquad\text{stable}\iff|z|<1 \]

\item left half $s$-plane $\to$ inside the unit circle
\item near $z=1$: slow; near $z=-1$: rings at $f_s/2$
\end{itemize}

\begin{frame}[fragile]{Discrete PID implementation}
\begin{columns}[T]
\begin{column}{.45\textwidth}
\small
Every $T$ seconds the controller runs a few lines:
\begin{itemize}\setlength\itemsep{.4mm}
\item \textbf{I}: add up the error: $I_k=I_{k-1}+K_iT\,e_k$
\item \textbf{D}: difference of the measurement, filtered
\item \textbf{saturation}: the output is limited (max.\ current)
\item \term{anti-windup}: when saturated, stop the integrator growing --- otherwise big overshoot later
\end{itemize}
\textbf{Picture:} windup = keeping your foot down on the accelerator while the car is already at full power.
\end{column}
\begin{column}{.53\textwidth}
\begin{notebox}
\scriptsize
\begin{semiverbatim}
// executed every T (ISR synced to PWM)
e  = r - y;
I += Ki*T*e;                    // integrate
D  = a*D - b*(y - y_old);       // filtered D on y
y_old = y;              // a=Tf/(Tf+T), b=Kd/(Tf+T)
u  = Kp*e + I + D;
if (u > UMAX)      \{ I -= u - UMAX; u = UMAX; \}
else if (u < -UMAX)\{ I -= u + UMAX; u = -UMAX;\}
write_pwm(u);       // takes effect next period
\end{semiverbatim}
\end{notebox}
{\scriptsize D acts on $-y$ so a step in $r$ does not cause a ``kick''.}
\end{column}
\end{columns}
\src{[{\AA}str\"om \& H\"agglund, 2006, ch.~3]; [Franklin, Powell \& Workman, 1998].}
\end{frame}

\begin{keybox}[Take-away]
A drive builds any voltage by switching very fast between on and off and averaging (PWM).
\end{keybox}

\begin{frame}{Gate drivers and dead time (preview)}
\begin{columns}[T]
\begin{column}{.43\textwidth}
\small
\textbf{Gate driver}: the ``muscle'' between the microcontroller and the power switch.
\begin{itemize}\setlength\itemsep{.3mm}
\item \term{isolation}: the top switch floats at hundreds of volts
\item fast gate current (several A)
\item short-circuit protection
\end{itemize}
\term{Dead time} $t_d$: a short pause with \emph{both} switches off at every change, to avoid \hl{shoot-through}.
\end{column}
\begin{column}{.55\textwidth}
\centering

\begin{frame}{Session 3.2 summary, key terms and check}
\begin{columns}[T]
\begin{column}{.5\textwidth}
\begin{keybox}[Key take-aways]
\begin{itemize}\setlength\itemsep{.3mm}
\item Rectifier $\to$ DC link $\to$ brake chopper $\to$ inverter $\to$ motor.
\item $V_{dc}\approx1.35V_{LL}$ = 540\,V at 400\,V mains.
\item Braking energy goes into $R_b$, not the capacitor.
\item Switches only on/off; dead time prevents shoot-through.
\end{itemize}
\end{keybox}
\begin{defbox}[Key terms]
\small
\term{DC link} capacitor bus $\cdot$ \term{leg} pair of switches $\cdot$ \term{duty cycle} on-time fraction $\cdot$ \term{shoot-through} both switches on
\end{defbox}
\end{column}
\begin{column}{.48\textwidth}
\begin{keybox}[Check your understanding]
\begin{enumerate}\setlength\itemsep{1.5mm}
\item Why can a servo drive not simply use a resistor-like ``half-on'' transistor to set the motor voltage?
\item A drive trips on DC-bus over-voltage only while braking. What is missing or too small?
\item In a cascade, which loop must be tuned first, and why?
\end{enumerate}
\end{keybox}

\begin{column}{.5\textwidth}
\begin{defbox}[Control]
\begin{itemize}\setlength\itemsep{.3mm}
\item filters cost phase; notches remove resonances; lead buys phase back
\item state feedback places poles; observers estimate states
\item robust vs.\ adaptive: one safe design vs.\ self-adjusting gains
\item sampling delay $\approx1.5T$ costs $360^\circ f_cT_d$
\end{itemize}
\end{defbox}
\end{column}
\begin{column}{.48\textwidth}
\begin{defbox}[Servo hardware]
\begin{itemize}\setlength\itemsep{.3mm}
\item drive + motor + transmission + encoder + controller
\item cascaded current/velocity/position loops
\item rectifier, DC link, brake chopper, inverter, dead time
\end{itemize}
\end{defbox}
\begin{keybox}[One sentence]
Servo bandwidth is limited by phase: filters, delay and resonances spend it; compensators and fast sampling buy it back.
\end{keybox}

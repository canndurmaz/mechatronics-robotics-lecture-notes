\section{Resonance, vibration and filter use (3.7)}

\begin{frame}{Refresher: natural frequency of a mass on a spring}
\begin{columns}[T]
\column{.42\textwidth}
\centering
\begin{tikzpicture}[font=\small]
  \fill[pattern=north east lines,pattern color=mkGray] (-1,3.2) rectangle (1,3.45);
  \draw[thick] (-1,3.2) -- (1,3.2);
  \draw[thick,mkOrange,decorate,decoration={coil,aspect=.6,segment length=2.5mm,amplitude=2.5mm}] (0,3.2) -- (0,1.6);
  \node[mkOrange,right] at (.35,2.4) {$k$};
  \draw[thick,fill=mkPale] (-.6,.8) rectangle (.6,1.6) node[midway]{$m$};
  \draw[{Latex}-{Latex},mkBlue,thick] (1.0,.7) -- (1.0,1.7) node[midway,right]{$x(t)$};
  % swing
  \begin{scope}[xshift=2.6cm]
    \draw[thick] (-.6,3.3) -- (.6,3.3);
    \draw[mkGray,dashed] (0,3.3) -- (0,1.0);
    \draw[thick,mkNavy] (0,3.3) -- (-.8,1.3); \draw[thick,mkNavy] (0,3.3) -- (.8,1.3);
    \fill[mkOrange] (.8,1.3) circle (.12); \fill[mkOrange!40] (-.8,1.3) circle (.12);
    \node[below] at (0,.9) {swing};
  \end{scope}
\end{tikzpicture}
\begin{itemize}\small
\item Pull and let go: it oscillates at its \term{natural frequency}.
\item A swing has one too: push in rhythm and it builds up.
\end{itemize}
\column{.55\textwidth}
\begin{eqbox}
\vspace{-2mm}
\[
\omega_n=\sqrt{\frac{k}{m}}\ \ [\si{rad/s}],\qquad f_n=\frac{\omega_n}{2\pi}\ \ [\si{Hz}]
\]
\vspace{-4mm}
\end{eqbox}
{\small $k$ spring stiffness [\si{N/m}], $m$ mass [\si{kg}]. Rotating version: $k$ in \si{N.m/rad}, $J$ in \si{kg.m^2}.}
\begin{keybox}[Example]
\small
1.\ $m=\SI{1}{kg}$, $k=\SI{100}{N/m}$\\
2.\ $\omega_n=\sqrt{100/1}=\SI{10}{rad/s}$\\
3.\ $f_n=10/(2\pi)\approx\SI{1.6}{Hz}$
\end{keybox}
\begin{itemize}\small
\item Stiffer spring $\Rightarrow$ higher $f_n$; heavier mass $\Rightarrow$ lower $f_n$.
\item \term{Resonance}: driving at $f_n$ gives a large response if damping is small.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Why mechanical resonance limits servo bandwidth}
\begin{columns}[T]
\column{.5\textwidth}
\begin{itemize}\small
\item Days 3--4 assumed a \term{rigid} load: motor and load move as one.
\item Real shafts, couplings, belts and ball screws are \term{compliant}: they behave like springs.
\item Inertia + spring = a mass--spring system with very little damping.
\item Raise the controller gain to make the axis stiffer and, at some point, the axis \hl{rings or becomes unstable}.
\end{itemize}
\column{.47\textwidth}
\centering

\begin{frame}{The two-mass model and its two special frequencies}
\begin{columns}[T]
\column{.47\textwidth}
\centering
\begin{tikzpicture}[font=\scriptsize,scale=.92]
  \draw[thick,fill=mkPale] (0,0) rectangle (1.2,1.2) node[midway]{$J_M$};
  \draw[thick,fill=mkCream] (3.4,0) rectangle (4.6,1.2) node[midway]{$J_L$};
  \draw[thick,mkOrange,decorate,decoration={coil,aspect=.6,segment length=2mm,amplitude=1.6mm}] (1.2,.6) -- (3.4,.6);
  \node[above] at (2.3,.8) {$k$};
  \draw[sig,mkOrange] (-.9,.6) -- (0,.6) node[pos=0,left]{$T_M$};
  \node[mkBlue] at (.6,1.5) {motor};
  \node[mkBlue] at (4.0,1.5) {load};
\end{tikzpicture}
\begin{eqbox}
\vspace{-2mm}
\[
\omega_\text{AR}=\sqrt{\frac{k}{J_L}},\qquad
\omega_\text{R}=\omega_\text{AR}\sqrt{1+R}
\]
\vspace{-5mm}
\[ R=J_L/J_M \ \text{(inertia ratio)} \]
\vspace{-4mm}
\end{eqbox}
\begin{itemize}\small
\item \term{Anti-resonance} (AR): load swings, motor stands still.
\item \term{Resonance} (R): motor and load swing against each other.
\end{itemize}
\column{.5\textwidth}
\begin{keybox}[Worked example]
\small
1.\ $J_M=\SI{1.2e-4}{kg.m^2}$, $J_L=\SI{6e-4}{kg.m^2}$, $k=\SI{600}{N.m/rad}$\\[1mm]
2.\ $\omega_\text{AR}=\sqrt{600/(6\cdot10^{-4})}=\SI{1000}{rad/s}$ $\Rightarrow f_\text{AR}=\SI{159}{Hz}$\\[1mm]
3.\ $R=6/1.2=5$, $\sqrt{1+R}=2.45$\\[1mm]
4.\ $\omega_\text{R}=\SI{2449}{rad/s}$ $\Rightarrow f_\text{R}=\SI{390}{Hz}$
\end{keybox}
\gofurther{Newton's law for each inertia with shaft torque $T_S=k(\theta_M-\theta_L)+c(\omega_M-\omega_L)$ gives the full transfer functions; a gear $i$ reflects $J_L/i^2$, $k/i^2$.}
\end{columns}
\end{frame}

\begin{frame}{Refresher: decibels (dB)}
\begin{columns}[T]
\column{.48\textwidth}
\begin{defbox}[Refresher: decibels]
\small
Gains span many orders of magnitude, so we use a logarithmic scale:
\[
G_\text{dB}=20\log_{10}|G|
\]
Multiplying gains $\Rightarrow$ \emph{adding} dB.
\end{defbox}
\begin{itemize}\small
\item 0 dB = gain 1 (signal passes unchanged).
\item Positive dB = amplified; negative dB = attenuated.
\item A \term{Bode plot}: gain in dB versus frequency on a log axis.
\end{itemize}
\column{.48\textwidth}
\centering
\renewcommand{\arraystretch}{1.2}
\small
\begin{tabular}{@{}c c@{}}
\toprule
\textbf{Gain $|G|$} & \textbf{dB}\\
\midrule
100 & $+40$\\
10 & $+20$\\
2 & $\approx +6$\\
1 & 0\\
0.5 & $\approx -6$\\
0.1 & $-20$\\
0.01 & $-40$\\
\bottomrule
\end{tabular}

\vspace{2mm}
\begin{keybox}[Example]
\small A resonance peak of $+12$ dB $\approx$ gain $\times4$ ($6+6$ dB $=2\times2$).
\end{keybox}
\end{columns}
\end{frame}

\addplot[mkSky,thick,dashed] table[x=f,y=p3]{\dat{notch}};
\end{groupplot}
\end{tikzpicture}
\column{.38\textwidth}
\begin{itemize}\small
\item A \term{notch} (band-stop) filter removes a narrow band around $f_n$ and passes everything else (0 dB).
\item Place $f_n$ on the resonance $f_\text{R}$.
\item \term{Depth}: how much it cuts. \term{Width}: how many Hz it covers.
\item Wide = robust if $f_\text{R}$ drifts, but costs more phase at low $f$.
\end{itemize}
\gofurther{the biquad form $N(s)=\dfrac{s^2+2\zeta_z\omega_n s+\omega_n^2}{s^2+2\zeta_p\omega_n s+\omega_n^2}$, depth $\zeta_z/\zeta_p$, and its Tustin discretisation.}
\end{columns}
\end{frame}

\begin{frame}{Setting up a notch: a practical recipe}
\begin{columns}[T]
\column{.52\textwidth}
\begin{enumerate}\small
\item Measure the frequency response at low gain (sine sweep).
\item Read $f_\text{R}$ from the peak; check it at different loads and positions.
\item Set the notch centre $f_n=f_\text{R}$, depth \SIrange{10}{20}{dB}.
\item Make it wider if $f_\text{R}$ moves by more than $\pm$\SI{10}{\%}.
\item Raise the velocity gain again and re-check the margins.
\end{enumerate}
\column{.45\textwidth}
\begin{keybox}[Rule of thumb]
\small Keep the notch at least 3--5 times above the loop bandwidth: $f_n\gtrsim(3$--$5)\,f_c$. Example: $f_c=\SI{50}{Hz}$, $f_n=\SI{390}{Hz}$ (ratio 7.8): only $\approx 9^\circ$ phase lost.
\end{keybox}
\begin{notebox}
\small If the resonance is too close to $f_c$: stiffen the mechanics or lower $f_c$; no filter can fix it.
\end{notebox}
\gofurther{an \term{adaptive notch} estimates $f_\text{R}$ online (e.g.\ FFT peak of the torque command) and moves $f_n$ when the payload changes.}
\end{columns}
\end{frame}

\begin{frame}{Summary and key terms: resonance and filters}
\begin{columns}[T]
\column{.52\textwidth}
\begin{itemize}\small
\item A flexible drive train is a mass--spring system with a sharp resonance.
\item $f_\text{AR}=\frac{1}{2\pi}\sqrt{k/J_L}$; $f_\text{R}$ is $\sqrt{1+R}$ times higher.
\item The resonance peak can push the loop gain above 0 dB $\Rightarrow$ ringing.
\item \hl{Notch} for one narrow peak; \hl{low-pass} for broad high-frequency noise.
\item Fix the mechanics first, filter second.
\end{itemize}
\column{.45\textwidth}
\begin{defbox}[Key terms]
\small
\term{natural frequency} rate of free oscillation\\
\term{compliance} springiness of a part\\
\term{inertia ratio} $R=J_L/J_M$\\
\term{dB} $20\log_{10}$ of a gain\\
\term{notch} narrow band-stop filter\\
\term{low-pass} passes slow, damps fast
\end{defbox}
{\scriptsize\color{mkGray} [Ellis, 2012]; [Franklin, Powell \& Emami-Naeini, 2019].}
\end{columns}
\end{frame}

\begin{defbox}[Refresher: sampling]
\small
A computer only sees the signal at instants $t=k\,T_s$; the \term{sampling frequency} is $f_s=1/T_s$. Like a film camera taking 24 pictures per second.
\end{defbox}
\begin{eqbox}

\vspace{-4mm}
\end{eqbox}
\begin{itemize}\small
\item Faster signals are not lost, they are \hl{disguised} as slower ones: \term{aliasing}.
\item Same as a wagon wheel in a film or a strobe light.
\end{itemize}

\end{itemize}

\end{keybox}

\begin{frame}{From an RC low-pass to one line of code}
\begin{columns}[T]
\column{.5\textwidth}
\centering
\begin{circuitikz}[scale=.75,transform shape]
  \draw (0,0) node[left]{$x$} to[short,o-] (.3,0) to[R,l=$R$] (2.6,0) to[short,-o] (3.6,0) node[right]{$y$};
  \draw (2.6,0) to[C,l=$C$] (2.6,-1.6) node[ground]{};
\end{circuitikz}
\begin{itemize}\small
\item You know the RC circuit: $y$ follows $x$ with time constant $\tau=RC$; corner $f_f=1/(2\pi\tau)$.
\item In software: each sample, move $y$ a fraction $\alpha$ of the way towards $x$:
\end{itemize}
\begin{eqbox}
\vspace{-2mm}
\[
y[k]=y[k-1]+\alpha\,\big(x[k]-y[k-1]\big),\quad \alpha\approx\frac{T_s}{\tau}
\]
\vspace{-4mm}
\end{eqbox}
\column{.47\textwidth}
\begin{keybox}[Worked example]
\small
1.\ Want $f_f=\SI{500}{Hz}$ $\Rightarrow$ $\tau=1/(2\pi\cdot500)=\SI{318}{\micro s}$\\[1mm]
2.\ $f_s=\SI{16}{kHz}$ $\Rightarrow$ $T_s=\SI{62.5}{\micro s}$\\[1mm]
3.\ $\alpha\approx 62.5/318=0.196$ (exact: $1-e^{-T_s/\tau}=0.178$)\\[1mm]
4.\ Code: \texttt{y += 0.178f*(x - y);}
\end{keybox}
\begin{notebox}
\small Every low-pass filter is also a \hl{delay} of about $\tau$ (here \SI{0.3}{ms}, 5 samples). Inside a loop that costs phase margin.
\end{notebox}
\end{columns}
\end{frame}

\begin{frame}{Moving average: the simplest digital filter}
\begin{columns}[T]
\column{.55\textwidth}
\centering
\begin{tikzpicture}
\begin{axis}[mkplot,width=\linewidth,height=5.3cm,xmin=0,xmax=0.5,ymin=-40,ymax=3,
  xlabel={frequency relative to sampling $f/f_s$},ylabel={gain [dB]},legend pos=north east]
\addplot[mkBlue,thick] table[x=f,y=N4]{\dat{mavg}};
\addplot[mkOrange,thick] table[x=f,y=N8]{\dat{mavg}};
\addplot[mkSky,thick] table[x=f,y=N16]{\dat{mavg}};
\legend{$N=4$,$N=8$,$N=16$}
\end{axis}
\end{tikzpicture}
\column{.43\textwidth}
\begin{eqbox}
\vspace{-2mm}
\[
y[k]=\frac1N\sum_{i=0}^{N-1}x[k-i]
\]
\vspace{-4mm}
\end{eqbox}
\begin{itemize}\small
\item Average of the last $N$ samples.
\item Cancels a disturbance \hl{exactly} if its period fits the window: zeros at $f=f_s/N,\,2f_s/N,\dots$
\item Delay: half the window, $\frac{N-1}{2}T_s$.
\item Weak damping between the zeros.
\end{itemize}
\begin{keybox}[Example]
\small $f_s=\SI{16}{kHz}$, $N=16$: cancels \SI{1}{kHz}, \SI{2}{kHz}, \dots
\end{keybox}
\end{columns}

\small A \term{packet} (or \term{frame}) is a short block of bits sent over a cable, like a letter: an envelope with addresses, the letter inside, and a seal that shows tampering.
\end{defbox}
\begin{itemize}\small
\item \term{Bit rate}: bits per second on the wire.
\item \term{Cycle time}: how often fresh packets are exchanged.
\item \term{Latency}: how old the data is when it arrives.
\item \term{Fieldbus}: a network built for machines, with guaranteed timing.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{CAN and CANopen}
\begin{columns}[T]
\column{.5\textwidth}
\begin{itemize}\small
\item \term{CAN}: two twisted wires shared by all nodes, $\SI{120}{\ohm}$ resistor at each end.
\item Each message has an \term{identifier}; if two nodes send at once, the lower identifier wins (\term{arbitration}) --- like a polite queue by priority.
\item Up to 8 data bytes per frame, \SI{1}{Mbit/s} over short cables.
\item \term{CANopen} adds a common dictionary of parameters and fast process-data messages (\term{PDO}).
\end{itemize}
\column{.47\textwidth}
\centering
\begin{tikzpicture}[font=\tiny,xscale=.95]
  \foreach \x/\w/\t/\c in {0/.35/SOF/mkGray,.35/1.3/ID 11 bit/mkBlue,1.65/.5/ctrl/mkMid,2.15/2.2/data 0--8 B/mkOrange,4.35/1.0/CRC 15/mkMid,5.35/.4/ACK/mkGray,5.75/.6/EOF/mkGray}{
    \draw[fill=\c!35,draw=\c] (\x,0) rectangle ++(\w,.5); \node at (\x+\w/2,.25) {\t};}
\end{tikzpicture}

\vspace{2mm}
\begin{keybox}[Worked example: how fast?]
\small
1.\ One full frame $\approx 120$ bits $\Rightarrow$ $\approx\SI{120}{\micro s}$ at \SI{1}{Mbit/s}\\
2.\ 4 axes $\times$ 2 frames + 1 sync frame $=9$ frames\\
3.\ $9\times\SI{120}{\micro s}\approx\SI{1.1}{ms}$ per cycle\\
$\Rightarrow$ fine for $\ge$\SI{1}{ms} cycles and few axes only.
\end{keybox}
\end{columns}
\end{frame}

\begin{frame}{CiA 402: a common language for drives}
\begin{columns}[T]
\column{.5\textwidth}
\centering
\begin{tikzpicture}[font=\scriptsize,node distance=3.5mm,
  st/.style={draw=mkBlue,fill=mkPale,rounded corners=2pt,minimum width=3.0cm,minimum height=5mm,align=center}]
  \node[st] (sd) {Switch on disabled};
  \node[st,below=of sd] (rs) {Ready to switch on};
  \node[st,below=of rs] (so) {Switched on};
  \node[st,below=of so,fill=mkCream,draw=mkOrange] (oe) {Operation enabled};
  \node[st,right=8mm of rs,minimum width=1.6cm,fill=mkRed!15,draw=mkRed] (f) {Fault};
  \draw[sig] (sd) -- (rs); \draw[sig] (rs) -- (so); \draw[sig] (so) -- (oe);
  \draw[sig,mkRed] (f) |- (sd) node[pos=.75,above]{reset};
\end{tikzpicture}
\begin{itemize}\small
\item Like a car: unlocked $\to$ ignition $\to$ engine on $\to$ driving. A fault drops it back.
\end{itemize}
\column{.47\textwidth}
\begin{itemize}\small
\item \term{CiA 402} is a \term{drive profile}: every vendor's drive understands the same commands and parameters.
\item The controller writes a \term{control word}, the drive answers with a \term{status word}.
\item Most used modes: \hl{cyclic synchronous position / velocity / torque} (CSP, CSV, CST): the controller sends a new set-point every bus cycle.
\item Same profile over CAN and EtherCAT.
\end{itemize}
\gofurther{the full state machine (8 states incl.\ quick stop and fault reaction), object indices 0x6040/0x6041, 0x607A, \dots, IEC 61800-7.}
\end{columns}
\end{frame}

\vspace{3mm}
\begin{columns}[T]
\column{.5\textwidth}
\begin{itemize}\small
\item Like a train passing stations: each drive \hl{unloads} its set-points and \hl{loads} its actual values while the frame rushes through (well under \SI{1}{\micro s} per drive).
\item A \term{working counter} in the frame tells the master that every drive answered.
\item \term{Distributed clocks}: all drives share one time base and act at the same instant.
\end{itemize}
\column{.47\textwidth}
\begin{keybox}[Example: 32 axes]
\small
1.\ $32\times\SI{12}{B}=\SI{384}{B}$ + \SI{52}{B} overhead\\
2.\ $\approx\SI{3500}{bit}$ at \SI{100}{Mbit/s} $\approx\SI{35}{\micro s}$\\
3.\ + delays $\approx\SI{30}{\micro s}$ $\Rightarrow$ fits a \SI{100}{\micro s} cycle
\end{keybox}
\includegraphics[height=1.0cm]{../figures/d5_ethercat_esc.jpg}\\
\credit{EtherCAT slave controller IC: Allanchou1357, CC BY-SA 4.0, via Wikimedia Commons}
\end{columns}
\vspace{1mm}
\gofurther{distributed-clock details: propagation-delay measurement, drift compensation, SYNC0 events, sub-\si{\micro s} jitter.}
\end{frame}

\begin{frame}{Latency budget: which loops may run over the bus?}
\centering
\begin{tikzpicture}[font=\scriptsize]
  \lbx{0}{1.2}{encoder}{mkSky!40}
  \lbx{1.2}{2.4}{wait\\for cycle}{mkPale}
  \lbx{2.4}{3.6}{bus to\\controller}{mkMid!25}
  \lbx{3.6}{5.4}{controller\\computes}{mkCream}
  \lbx{5.4}{6.6}{bus to\\drive}{mkMid!25}
  \lbx{6.6}{7.8}{apply}{mkPale}
  \lbx{7.8}{9.4}{current\\loop}{mkSky!40}
  \draw[{Latex}-{Latex},mkOrange,thick] (0,-.25) -- (9.4,-.25) node[midway,below]{total delay $T_d\approx 2$--$3$ bus cycles};
\end{tikzpicture}

\vspace{3mm}
\begin{columns}[T]
\column{.5\textwidth}
\begin{keybox}[Worked example]
\small
1.\ Bus cycle \SI{250}{\micro s}, $T_d=2.5\times250=\SI{625}{\micro s}$\\[1mm]
2.\ Position loop, $f_c=\SI{30}{Hz}$: $360\cdot30\cdot625\cdot10^{-6}=7^\circ$ \textcolor{mkGreen}{ok}\\[1mm]
3.\ Velocity loop, $f_c=\SI{200}{Hz}$: $45^\circ$ \textcolor{mkRed}{too much}
\end{keybox}
\column{.47\textwidth}
\begin{itemize}\small
\item Close the fast loops (current, velocity) \hl{inside the drive}.
\item Send only position set-points over the bus (CSP mode).
\item Shower analogy: a long pipe (delay) makes temperature control slow --- keep fast control close to the tap.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Is the data valid? CRC, counters, watchdogs, timestamps}
\begin{columns}[T]
\column{.5\textwidth}
\begin{itemize}\small
\item \term{CRC}: detects bits damaged on the way. But a correct value can still be \emph{old}.
\item \term{Counters}: a number that increases every cycle; a frozen counter = lost data.
\item \term{Watchdog}: a timer that trips if no fresh data arrives in time.
\item \term{Timestamp}: age of a value $=t_\text{now}-t_\text{sample}$.
\item \term{Plausibility}: is the value possible? e.g.\ $i_a+i_b+i_c=0$.
\end{itemize}
\column{.47\textwidth}

\begin{frame}{Summary and key terms: communication and integration}
\begin{columns}[T]
\column{.52\textwidth}
\begin{itemize}\small
\item Packets = header + data + CRC; buses exchange them every cycle.
\item CAN/CANopen: robust, priority arbitration, $\ge$\SI{1}{ms} cycles.
\item CiA 402: one command language for all drives.
\item EtherCAT: one frame through all drives, \SI{100}{\micro s} cycles, common clock.
\item Delay costs phase: close fast loops inside the drive; check freshness of data.
\end{itemize}
\column{.45\textwidth}
\begin{defbox}[Key terms]
\small
\term{packet} header + data + check\\
\term{fieldbus} machine network with fixed timing\\
\term{arbitration} who may send first\\
\term{drive profile} standard drive commands\\
\term{latency} age of data on arrival\\
\term{watchdog} timer that trips on missing data
\end{defbox}
{\scriptsize\color{mkGray} ETG.1000 / IEC 61158; CiA 301, CiA 402; ISO 11898.}
\end{columns}
\end{frame}

\begin{frame}{Course wrap-up: ten principles to take home}
\begin{columns}[T]
\column{.5\textwidth}
\begin{enumerate}\small
\item Know your plant: motor \emph{and} mechanics (Days 1, 5).
\item Model first, then measure and check (Day 2).
\item Feedback needs margins: keep a safe phase margin (Day 2).
\item Every filter and every sample adds delay (Days 3, 5).
\item Cascade loops: inner loops 3--10$\times$ faster (Day 4).
\end{enumerate}
\column{.48\textwidth}
\begin{enumerate}\small
\setcounter{enumi}{5}
\item Field orientation makes the motor a torque source (Day 4).
\item Feed-forward does the work, feedback fixes errors (Day 4).
\item Fix resonances mechanically first, then notch / LPF (Day 5).
\item Sensor resolution, noise and delay limit stiffness (Day 5).
\item Synchronise, time-stamp and check all data (Day 5).
\end{enumerate}
\end{columns}

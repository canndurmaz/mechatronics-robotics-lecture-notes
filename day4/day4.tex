\begin{frame}[t]{Refresher: duty cycle and average voltage}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\centering
\begin{tikzpicture}[x=.95cm,y=1.0cm]
  \draw[-{Latex[length=2mm]},mkGray] (0,0) -- (6.3,0) node[right,font=\scriptsize]{$t$};

  \foreach \k in {0,1,2}{
    \draw[mkBlue,line width=1pt] (2*\k,0) -- (2*\k,2) -- (2*\k+.8,2) -- (2*\k+.8,0) -- (2*\k+2,0);
  }
  \draw[mkOrange,line width=1.1pt,dashed] (0,.8) -- (6,.8) node[right,font=\scriptsize]{$\bar v=D\,V_\text{dc}$};
  \node[font=\scriptsize,anchor=east] at (0,2) {$V_\text{dc}$};
  \draw[|-|,mkMid] (0,2.25) -- node[above,font=\tiny]{$t_\text{on}$} (.8,2.25);
  \draw[|-|,mkGray] (2,-.35) -- node[below,font=\tiny]{period $T$} (4,-.35);
\end{tikzpicture}\\[2mm]
\small Analogy: flick a light switch on and off very fast -- the room looks evenly \emph{dimmed}.
The motor's inductance does the same for current: like a flywheel, it smooths the pulses.
\column{.46\textwidth}
\begin{itemize}\small
\item \term{PWM} (pulse-width modulation): switch fully on/off at a fixed rate, vary the on-time.
\item \term{Duty cycle} $D=t_\text{on}/T$ (0 \ldots 1).
\end{itemize}
\begin{eqbox}\centering
$\bar v = D\cdot V_\text{dc}$
\end{eqbox}
\begin{defbox}[Example]
\small $V_\text{dc}=\SI{325}{V}$, $T=\SI{62.5}{\micro s}$ (\SI{16}{kHz}), $t_\text{on}=\SI{25}{\micro s}$:\\
$D=25/62.5=0.4$ $\Rightarrow$ $\bar v=0.4\cdot\SI{325}{V}=\SI{130}{V}$.
\end{defbox}
\small\hl{Take-away:} slowly changing $D$ gives a slowly changing average voltage -- e.g.\ a sine.
\end{columns}
\end{frame}

\begin{columns}[T,onlytextwidth]

\begin{tikzpicture}
\begin{axis}[mkplot,width=6.2cm,height=3.0cm,xmin=-1,xmax=1,ymin=-1.4,ymax=1.4,xtick={0},ytick={-1,0,1},

\column{.48\textwidth}
\begin{itemize}\small
\item Both switches of a leg ON together = short circuit of the DC link (\term{shoot-through}).
\item So each turn-on waits a short \term{dead time} $t_d$ (0.5--3\,\si{\micro s}).
\item During $t_d$ a diode conducts: the voltage depends on the \hl{current direction}, not on the command.
\end{itemize}
\begin{eqbox}\centering
$\Delta \bar V\approx t_d\,f_\text{sw}\,V_\text{dc}$
\end{eqbox}
\begin{defbox}[Example]
\small $t_d=\SI{1}{\micro s}$, $f_\text{sw}=\SI{16}{kHz}$, $V_\text{dc}=\SI{325}{V}$:\\
$\Delta \bar V=10^{-6}\cdot16000\cdot325=\SI{5.2}{V}$ -- as big as the whole back-EMF at \SI{100}{rpm}!
\end{defbox}
\small Cure: dead-time compensation + the current loop.
\end{columns}
\end{frame}

\end{defbox}

\begin{frame}{Summary 3.2 -- PWM and limits}
\begin{columns}[T,onlytextwidth]
\column{.55\textwidth}
\begin{keybox}[Key take-aways]
\small
\begin{itemize}
\item PWM: average voltage $=D\cdot V_\text{dc}$; a sinusoidal duty cycle gives a sinusoidal voltage.
\item Sine PWM reaches $V_\text{dc}/2$, SVPWM $V_\text{dc}/\sqrt3$ (+15\,\%).
\item Dead time prevents shoot-through but adds a voltage error.
\item Higher $f_\text{sw}$: less ripple and delay, more loss.
\item Current limit $\to$ max.\ torque; voltage limit $\to$ max.\ speed.
\end{itemize}
\end{keybox}
\column{.42\textwidth}
\begin{defbox}[Key terms]
\small
\textbf{Duty cycle} -- fraction of time ON.\\
\textbf{PWM} -- pulse-width modulation.\\
\textbf{Modulation index $m$} -- sine amplitude / max.\\
\textbf{SVPWM} -- space-vector PWM.\\
\textbf{Dead time} -- both switches OFF briefly.\\
\textbf{Base speed} -- where the voltage runs out.
\end{defbox}
\end{columns}
\end{frame}

\section{3.3 Field-Oriented Control (FOC)}
% =====================================================================

\begin{frame}[t]{Why field-oriented control? The DC-motor analogy}
\begin{columns}[T,onlytextwidth]
\column{.58\textwidth}
\begin{itemize}\small
\item Brushed DC motor: the commutator keeps the current at $90^\circ$ to the magnet $\Rightarrow$ simply $T=k_t\,i$.
\item PM synchronous motor (\term{PMSM}): the magnet is on the rotor and turns -- the current must turn with it.
\item Analogy: to open a door with least effort, push \hl{perpendicular} to the door -- even while it swings.
\item \term{FOC} = electronic commutator: measure the rotor angle $\theta_e$ and split the current into
\begin{itemize}\small
\item $i_d$ -- along the magnet (no torque),
\item $i_q$ -- $90^\circ$ ahead (makes torque).
\end{itemize}
\end{itemize}
\src{[Blaschke, 1972]; [Leonhard, 2001]; [Krishnan, 2010]}
\column{.40\textwidth}
\centering
\begin{tikzpicture}[scale=.95]

  \draw[-{Latex[length=2mm]},mkBlue,line width=1.1pt] (0,0) -- (15:2) node[right,font=\scriptsize]{$\vec u$};
  \draw[-{Latex[length=2mm]},mkOrange,line width=1.1pt] (0,0) -- (65:2) node[above,font=\scriptsize]{$\vec u'=R(\theta)\,\vec u$};
  \draw[mkOrange,-{Latex[length=1.5mm]}] (15:.9) arc (15:65:.9); \node[font=\scriptsize,mkOrange] at (40:1.1) {$\theta$};
  \draw[mkGray,dashed] (0,0) circle (2);
\end{tikzpicture}\\
\small Length stays the same, only the direction changes.
\column{.56\textwidth}
\begin{itemize}\small
\item A vector $(x,y)$ = the complex number $x+jy$. Multiplying by $e^{j\theta}$ \hl{rotates} it by $\theta$.
\item The same in matrix form:
\end{itemize}
\begin{eqbox}\centering
$R(\theta)=\begin{bmatrix}\cos\theta&-\sin\theta\\ \sin\theta&\cos\theta\end{bmatrix}$
\end{eqbox}
\begin{defbox}[Example]
\small Rotate $(1,0)$ by $90^\circ$: $\begin{bmatrix}0&-1\\1&0\end{bmatrix}\!\begin{bmatrix}1\\0\end{bmatrix}=\begin{bmatrix}0\\1\end{bmatrix}$.\\[1mm]
Turning the \emph{axes} by $+\theta$ instead = turning the vector by $-\theta$: $(3,4)$ seen from axes turned $90^\circ$ reads $(4,-3)$.
\end{defbox}
\small\hl{Take-away:} $R(-\theta)$ = ``what the vector looks like to a turned observer''.
\end{columns}
\end{frame}

\begin{frame}[t]{Clarke transform: three phases $\to$ two axes ($\alpha\beta$)}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{defbox}[Clarke: $abc\to\alpha\beta$ (fixed to the stator)]
\vspace{-2mm}
\[\begin{bmatrix} i_\alpha\\ i_\beta\end{bmatrix}=\frac23\begin{bmatrix}1&-\tfrac12&-\tfrac12\\[1pt] 0&\tfrac{\sqrt3}{2}&-\tfrac{\sqrt3}{2}\end{bmatrix}\begin{bmatrix} i_a\\ i_b\\ i_c\end{bmatrix}\]
\end{defbox}
\begin{itemize}\small
\item Each phase current pushes along its own coil axis; add the three pushes as arrows, write the result in $x$--$y$ form.
\item Because $i_a+i_b+i_c=0$: $i_\alpha=i_a$,\ $i_\beta=(i_a+2i_b)/\sqrt3$ -- two current sensors suffice.
\item Factor $\tfrac23$: arrow length = peak phase current $\hat I$.
\end{itemize}
\column{.46\textwidth}
\centering
\begin{tikzpicture}[scale=1.25]
  \foreach \a/\n in {0/a,120/b,240/c}{\draw[mkGray!70,dashed] (0,0) -- (\a:1.2); \node[font=\scriptsize,mkGray] at (\a:1.35) {$\n$};}
  \draw[-{Latex[length=1.6mm]},mkBlue,line width=.9pt] (0,0) -- (1.333,0) node[midway,above,font=\tiny]{$\tfrac23 i_a$};
  \draw[-{Latex[length=1.6mm]},mkOrange,line width=.9pt] (1.333,0) -- (1.667,-.577) node[midway,left,font=\tiny]{$\tfrac23 i_b$};
  \draw[-{Latex[length=1.6mm]},mkSky,line width=.9pt] (1.667,-.577) -- (2,0) node[midway,right,font=\tiny]{$\tfrac23 i_c$};
  \draw[-{Latex[length=2mm]},mkMid,line width=1.3pt] (0,-.9) -- (2,-.9) node[right,font=\scriptsize]{$\vec i$};
  \node[font=\scriptsize,mkMid] at (1,-1.15) {$i_\alpha=8,\ i_\beta=0$};
\end{tikzpicture}\\[2mm]
\begin{defbox}[Example]
\small $i_a=\SI{8}{A}$, $i_b=i_c=\SI{-4}{A}$:\\
$i_\alpha=\tfrac23(8+2+2)=\SI{8}{A}$,\\ $i_\beta=(8-8)/\sqrt3=\SI{0}{A}$.
\end{defbox}
\small The arrow still \hl{rotates} at $\omega_e$: $i_\alpha$, $i_\beta$ are still sines [Clarke, 1943].
\end{columns}
\end{frame}

\begin{frame}[t]{Current-loop PI: a step-by-step recipe}
\begin{columns}[T,onlytextwidth]
\column{.50\textwidth}
\begin{itemize}\small
\item Plant (one axis): the $R$--$L$ winding, $G_i(s)=\dfrac{1}{R+Ls}$, plus a small \hl{delay} $T_\sigma$ (compute + PWM).
\item PI: $C_i(s)=K_p\dfrac{1+sT_i}{sT_i}$.
\item Idea: let the PI's zero cancel the slow winding pole, then pick $K_p$ so the delay is still safe.
\end{itemize}
\begin{eqbox}\centering
$T_i=\dfrac{L}{R},\qquad K_p=\dfrac{L}{2\,T_\sigma}$
\end{eqbox}
\small This rule is called the \term{technical optimum} (TO).
\gofurther{the closed loop becomes 2nd order with $\zeta=0.707$ (4.3\,\% overshoot); saturation lowers $L$ at high current, so check $K_p$ there.}
\column{.48\textwidth}
\begin{defbox}[Worked example (reference motor)]
\small
\begin{enumerate}
\item $T_s=1/\SI{16}{kHz}=\SI{62.5}{\micro s}$; $T_\sigma=1.5\,T_s=\SI{93.75}{\micro s}$.
\item $T_i=\SI{3}{mH}/\SI{0.5}{\ohm}=\SI{6}{ms}$.
\item $K_p=\dfrac{\SI{3}{mH}}{2\cdot\SI{93.75}{\micro s}}=\SI{16}{V/A}$.
\item Crossover $\omega_c\approx K_p/L=\SI{5333}{rad/s}$ ($f_c\approx\SI{850}{Hz}$); delay costs $360^\circ\cdot850\cdot93.75\,\si{\micro s}=29^\circ$ $\Rightarrow$ PM $\approx61^\circ$.
\end{enumerate}
\end{defbox}
\end{columns}
\end{frame}

\begin{frame}[t]{Velocity-loop PI: a step-by-step recipe}
\begin{columns}[T,onlytextwidth]
\column{.50\textwidth}
\begin{itemize}\small
\item Plant: current $\to$ torque $\to$ speed. Newton: $J\dot\omega=k_ti_q$ $\Rightarrow$ an \hl{integrator} $\dfrac{k_t}{Js}$.
\item Plus a small lag $T_\text{eq}$: closed current loop + speed filter.
\item Choose a spacing factor $a$ (2\ldots4): crossover $a$ times below the lag, PI corner $a$ times below the crossover.
\end{itemize}
\begin{eqbox}\centering
$K_{p\omega}=\dfrac{J\,\omega_c}{k_t},\qquad \omega_c=\dfrac{1}{a\,T_\text{eq}}$
\end{eqbox}
\gofurther{this is the \term{symmetric optimum} [Kessler, 1958]: $T_{i\omega}=a^2T_\text{eq}$, PM $=\arctan\frac{a^2-1}{2a}$ (37$^\circ$ at $a=2$, 53$^\circ$ at $a=3$); a reference prefilter removes the large step overshoot.}
\column{.48\textwidth}
\begin{defbox}[Worked example: spacing factor 2]
\small
\begin{enumerate}
\item $T_\text{eq}=2\cdot\SI{93.75}{\micro s}+\SI{0.31}{ms}\approx\SI{0.5}{ms}$.
\item $\omega_c=1/(2\cdot\SI{0.5}{ms})=\SI{1000}{rad/s}$ (\SI{159}{Hz}).
\item $J=J_m(1+3)=1.2\cdot10^{-3}$\,kg\,m$^2$; $K_{p\omega}=1.2\cdot10^{-3}\cdot1000/0.72=1.67$\,A\,s/rad.
\item $T_{i\omega}=a^2T_\text{eq}=\SI{2}{ms}$.
\end{enumerate}
\end{defbox}
\small Larger $a$ $\Rightarrow$ slower but calmer ($a=3$: $f_c=\SI{106}{Hz}$).
\end{columns}
\end{frame}

\begin{frame}[t]{Position loop: a P controller and the following error}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{itemize}\small
\item With a good velocity loop, speed $\approx$ command; position is the integral of speed.
\item So a simple \hl{P controller} reaches any fixed target with zero error. Gain $k_v$ (1/s).
\item Following a ramp (constant speed $v$) leaves a constant lag: the \term{following error}.
\end{itemize}
\begin{eqbox}\centering
$e_\text{ss}=\dfrac{v}{k_v}$\qquad (P loop, no feedforward)
\end{eqbox}
\small Analogy: a dog on a leash trotting behind its owner -- the faster the walk, the longer the leash.\\[1mm]
Rule: $k_v\approx$ velocity crossover / 4\ldots10. No overshoot allowed in machining.
\column{.46\textwidth}
\begin{defbox}[Worked example]
\small
\begin{enumerate}
\item Machine-tool axis: $k_v=\SI{50}{s^{-1}}$, feed $v=\SI{10}{m/min}=\SI{167}{mm/s}$.
\item $e_\text{ss}=167/50=\SI{3.3}{mm}$ (!)
\item Doubling $k_v$ halves it -- but stability limits $k_v$.
\end{enumerate}
\end{defbox}
\begin{notebox}
Two axes cutting a contour need \hl{equal} $k_v$, otherwise the path is distorted -- or use feedforward (two slides on).
\end{notebox}
\end{columns}
\end{frame}

\begin{frame}[t]{Torque and speed limits; integrator windup}
\begin{columns}[T,onlytextwidth]
\column{.45\textwidth}
\begin{itemize}\small
\item Limits: $|i_q^*|\le\hat I_\text{max}$, $|\omega^*|\le\omega_\text{max}$, plus the voltage limit.
\item While the output is limited, the loop is effectively \hl{open}; the integrator keeps adding up error.
\item Result: \term{windup} -- huge overshoot when the limit releases.
\item Analogy: holding the accelerator floored and still ``remembering'' to push harder.
\item Cure (\term{anti-windup}): stop integrating while saturated (clamping).
\end{itemize}

\begin{frame}{Summary 3.4 -- Cascade loops}
\begin{columns}[T,onlytextwidth]
\column{.56\textwidth}
\begin{keybox}[Key take-aways]
\small
\begin{itemize}
\item Cascade: position (P) $\to$ velocity (PI) $\to$ current (PI); inner = faster.
\item Current PI: $T_i=L/R$, $K_p=L/(2T_\sigma)$ $\Rightarrow$ $\approx\SI{850}{Hz}$.
\item Velocity PI: $K_p=J\omega_c/k_t$ with $\omega_c=1/(aT_\text{eq})$.
\item P position loop: following error $v/k_v$.
\item Limits need anti-windup; S-curves limit jerk; feedforward removes most following error.
\end{itemize}
\end{keybox}
\column{.42\textwidth}
\begin{defbox}[Key terms]
\small
\textbf{Cascade} -- loops nested inside each other.\\
\textbf{Crossover $\omega_c$} -- where loop gain $=1$.\\
\textbf{$k_v$} -- position-loop gain (1/s).\\
\textbf{Following error} -- lag behind a moving target.\\
\textbf{Windup} -- integrator overfills during a limit.\\
\textbf{Jerk} -- rate of change of acceleration.\\
\textbf{Feedforward} -- act on the known command.
\end{defbox}
\end{columns}

\item Outside the loop: no stability change.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}[t]{Disturbance observer (DOB): estimate the unknown push}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{tikzpicture}
\begin{groupplot}[group style={group size=1 by 2,vertical sep=3mm,xticklabels at=edge bottom},
  mkplot,width=7.8cm,xmin=0,xmax=30,legend pos=south east]
\nextgroupplot[height=3.6cm,ylabel={$\omega$ (rad/s)},ymin=-2.1,ymax=.4]
\addplot[mkRed,thick] table[x=t,y=w_pi]{data/dob.dat};
\addplot[mkBlue,thick] table[x=t,y=w_dob]{data/dob.dat};
\legend{PI only,PI + DOB}
\nextgroupplot[height=2.8cm,ylabel={$\hat T_L$ (Nm)},xlabel={time (ms)},ymin=-.2,ymax=2.4]
\addplot[mkOrange,thick] table[x=t,y=dhat]{data/dob.dat};
\addplot[mkGray,dashed] coordinates {(0,0) (5,0) (5,2) (30,2)};
\end{groupplot}
\end{tikzpicture}\\
\scriptsize Simulated: holding $\omega^*=0$, load step \SI{2}{Nm} at \SI{5}{ms}.
\column{.46\textwidth}
\begin{itemize}\small
\item Idea: compare what the motor \emph{should} do ($k_ti_q$) with what it \emph{does} ($\hat J\dot\omega$). The difference is the unknown disturbance.
\end{itemize}
\begin{eqbox}\centering
$\hat T_L\approx k_t\,i_q-\hat J\,\dot\omega$\ \ (low-pass filtered)
\end{eqbox}
\begin{itemize}\small
\item Add $\hat T_L/k_t$ to the current command $\Rightarrow$ the disturbance is cancelled.
\item Result: speed dip \SI{1.85}{rad/s} $\to$ \SI{0.70}{rad/s} (2.6$\times$ smaller).
\item Bonus: $\hat T_L$ can detect collisions.
\end{itemize}
\gofurther{the filter $Q(s)=1/(1+s\tau_Q)$ makes the derivative usable; $1/\tau_Q\approx$ 0.5--2$\times\omega_c$, limited by noise and delay [Ohnishi, Shibata \& Murakami, 1996].}
\end{columns}
\end{frame}

\end{axis}
\end{tikzpicture}\\
\scriptsize Illustrative 12-slot/8-pole motor: \SI{0.146}{Nm} $\to$ \SI{0.015}{Nm} peak-to-peak.
\column{.43\textwidth}
\begin{itemize}\small
\item \term{Cogging}: magnets ``click'' into the slots -- felt when you turn an unpowered motor by hand.
\item Repeats every $360^\circ/\text{LCM}(\text{slots},\text{poles})$.
\end{itemize}
\begin{defbox}[Example]
\small 12 slots, 8 poles: LCM $=24$ $\Rightarrow$ period $360^\circ/24=15^\circ$ mech.
\end{defbox}
\begin{itemize}\small
\item Cure: measure $i_q$ vs.\ angle at slow speed, store in a \term{look-up table}, add it to $i_q^*$.
\item Matters most at low speed.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Summary 3.6 -- Disturbances and compensation}
\begin{columns}[T,onlytextwidth]
\column{.56\textwidth}
\begin{keybox}[Key take-aways]
\small
\begin{itemize}
\item Load torque and friction push against the motor; PI fixes them slowly.
\item Feedforward cancels \emph{known} effects from the plan.
\item A DOB estimates the \emph{unknown} push from $k_ti_q-\hat J\dot\omega$.
\item Cogging repeats with angle $\Rightarrow$ table compensation.
\end{itemize}
\end{keybox}
\column{.42\textwidth}
\begin{defbox}[Key terms]
\small
\textbf{Load torque $T_L$} -- external push on the shaft.\\
\textbf{Stick-slip} -- jerky motion from friction.\\
\textbf{DOB} -- disturbance observer.\\
\textbf{Cogging} -- magnet--slot torque at zero current.\\
\textbf{Look-up table} -- stored correction vs.\ angle.
\end{defbox}
